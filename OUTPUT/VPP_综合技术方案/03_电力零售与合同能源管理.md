```python
from enum import Enum
from datetime import datetime, timedelta
from typing import List, Dict, Optional, Callable

class ContractType(Enum):
    """合同类型枚举"""
    RETAIL_FIXED = "零售固定电价"
    RETAIL_TOU = "零售分时电价"
    RETAIL_REAL_TIME = "零售实时电价"
    PPA_CORPORATE = "企业购电协议(PPA)"
    EMC_SHARED_SAVINGS = "EMC效益分享型"
    EMC_GUARANTEED_SAVINGS = "EMC节电保证型"
    VPP_AGGREGATION = "VPP聚合协议"
    GREEN_CERTIFICATE = "绿证交易合同"

class PriceModel(Enum):
    """电价模型枚举"""
    FIXED = "fixed"  # 固定电价
    TOU = "tou"  # 分时电价
    RTP = "rtp"  # 实时电价
    CPP = "cpp"  # 尖峰电价
    BLOCK = "block"  # 阶梯电价
    INDEXED = "indexed"  # 指数挂钩


class ContractMeta:
    """
    合同元数据模型
    将合同条款转化为可计算的数学函数
    """
    def __init__(self, contract_id: str, contract_type: ContractType,
                 start_date: datetime, end_date: datetime,
                 counterparty: str, capacity_kw: float):
        self.contract_id = contract_id
        self.contract_type = contract_type
        self.start_date = start_date
        self.end_date = end_date
        self.counterparty = counterparty
        self.capacity_kw = capacity_kw

        # 价格条款
        self.price_model: PriceModel = PriceModel.FIXED
        self.price_functions: List[PriceFunction] = []

        # 分成条款
        self.share_functions: List[ShareFunction] = []

        # 保底条款
        self.guarantee_functions: List[GuaranteeFunction] = []

        # 违约条款
        self.penalty_functions: List[PenaltyFunction] = []

        # 版本控制
        self.version = 1
        self.modified_at = datetime.now()

    def add_price_function(self, price_func: 'PriceFunction'):
        """添加电价函数"""
        self.price_functions.append(price_func)

    def add_share_function(self, share_func: 'ShareFunction'):
        """添加分成函数"""
        self.share_functions.append(share_func)

    def add_guarantee_function(self, guarantee_func: 'GuaranteeFunction'):
        """添加保底函数"""
        self.guarantee_functions.append(guarantee_func)

    def add_penalty_function(self, penalty_func: 'PenaltyFunction'):
        """添加违约函数"""
        self.penalty_functions.append(penalty_func)

    def calculate_price(self, time_point: datetime, base_price: float) -> float:
        """
        计算指定时间的合同电价
        """
        price = base_price
        for func in self.price_functions:
            price = func.apply(time_point, price)
        return price

    def calculate_settlement(self, period_start: datetime, period_end: datetime,
                              energy_kwh: float, market_price: float) -> Dict:
        """
        计算结算金额
        """
        # 1. 计算合同电价
        contract_price = self.calculate_price(period_start, market_price)

        # 2. 计算基础电费
        base_amount = energy_kwh * contract_price

        # 3. 计算分成
        share_amount = 0
        for func in self.share_functions:
            share_amount += func.apply(base_amount)

        # 4. 检查保底
        guarantee_amount = 0
        for func in self.guarantee_functions:
            guarantee_amount = func.apply(base_amount)

        # 5. 最终结算金额
        final_amount = max(base_amount + share_amount, guarantee_amount)

        return {
            "contract_id": self.contract_id,
            "period": f"{period_start}/{period_end}",
            "energy_kwh": energy_kwh,
            "contract_price": contract_price,
            "market_price": market_price,
            "base_amount": base_amount,
            "share_amount": share_amount,
            "guarantee_amount": guarantee_amount,
            "final_amount": final_amount
        }
```

```python
class PriceFunction:
    """
    电价函数基类
    将合同中的电价条款转化为数学函数
    """
    def apply(self, time_point: datetime, base_price: float) -> float:
        raise NotImplementedError


class FixedPriceFunction(PriceFunction):
    """固定电价函数"""
    def __init__(self, price_per_kwh: float):
        self.price = price_per_kwh

    def apply(self, time_point: datetime, base_price: float) -> float:
        return self.price


class TOUPriceFunction(PriceFunction):
    """
    分时电价函数
    支持峰/谷/平三段式电价
    """
    def __init__(self, peak_price: float, valley_price: float,
                 flat_price: float, peak_hours: List[int],
                 valley_hours: List[int]):
        self.peak_price = peak_price
        self.valley_price = valley_price
        self.flat_price = flat_price
        self.peak_hours = set(peak_hours)
        self.valley_hours = set(valley_hours)

    def apply(self, time_point: datetime, base_price: float) -> float:
        hour = time_point.hour
        if hour in self.peak_hours:
            return self.peak_price
        elif hour in self.valley_hours:
            return self.valley_price
        else:
            return self.flat_price


class IndexedPriceFunction(PriceFunction):
    """
    指数挂钩电价函数
    电价 = 基准价 + 指数调整
    """
    def __init__(self, base_price: float, index_name: str,
                 coefficient: float, floor_price: float = 0,
                 ceiling_price: float = float('inf')):
        self.base_price = base_price
        self.index_name = index_name  # 例如 "CPI", "PPI", "煤炭价格指数"
        self.coefficient = coefficient
        self.floor = floor_price
        self.ceiling = ceiling_price

    def apply(self, time_point: datetime, base_price: float) -> float:
        # 实际中从指数数据源获取
        index_value = self._get_index_value(time_point)
        adjusted_price = self.base_price + self.coefficient * index_value
        return max(self.floor, min(self.ceiling, adjusted_price))

    def _get_index_value(self, time_point: datetime) -> float:
        """获取指数值（实际实现中调用外部API）"""
        return 0.0  # 简化示例


class SichuanSeasonalPriceFunction(PriceFunction):
    """
    四川丰枯水期差异电价函数
    丰水期（6-10月）：电价下浮
    枯水期（12-4月）：电价上浮
    平水期（5月、11月）：基准电价
    """
    def __init__(self, base_price: float, wet_season_factor: float = 0.85,
                 dry_season_factor: float = 1.15):
        self.base_price = base_price
        self.wet_factor = wet_season_factor  # 丰水期下浮系数
        self.dry_factor = dry_season_factor  # 枯水期上浮系数

    def apply(self, time_point: datetime, base_price: float) -> float:
        month = time_point.month

        # 丰水期：6-10月
        if 6 <= month <= 10:
            return self.base_price * self.wet_factor
        # 枯水期：12-4月
        elif month == 12 or month <= 4:
            return self.base_price * self.dry_factor
        # 平水期：5月、11月
        else:
            return self.base_price


class PeakShavingPriceFunction(PriceFunction):
    """
    尖峰电价函数
    在特定尖峰时段执行高电价
    """
    def __init__(self, normal_price: float, peak_price: float,
                 peak_periods: List[Dict]):
        """
        peak_periods: [{"start_hour": 14, "end_hour": 16, "days": ["weekday"]}]
        """
        self.normal_price = normal_price
        self.peak_price = peak_price
        self.peak_periods = peak_periods

    def apply(self, time_point: datetime, base_price: float) -> float:
        for period in self.peak_periods:
            if self._is_in_period(time_point, period):
                return self.peak_price
        return self.normal_price

    def _is_in_period(self, time_point: datetime, period: Dict) -> bool:
        """判断是否在尖峰时段内"""
        hour = time_point.hour
        weekday = time_point.weekday()

        # 检查日期类型
        day_type = period.get("days", ["all"])
        if "weekday" in day_type and weekday >= 5:
            return False
        if "weekend" in day_type and weekday < 5:
            return False

        # 检查小时范围
        start = period["start_hour"]
        end = period["end_hour"]
        return start <= hour < end
```

### 1.4 分成与保底函数

```python
class ShareFunction:
    """
    分成函数基类
    将合同中的分成比例转化为数学函数
    """
    def apply(self, base_amount: float) -> float:
        raise NotImplementedError


class FixedRatioShareFunction(ShareFunction):
    """固定比例分成函数"""
    def __init__(self, ratio: float, payer: str = "VPP"):
        """
        ratio: 分成比例（0-1）
        payer: "VPP" 或 "CLIENT"
        """
        self.ratio = ratio
        self.payer = payer

    def apply(self, base_amount: float) -> float:
        share = base_amount * self.ratio
        return -share if self.payer == "VPP" else share


class TieredShareFunction(ShareFunction):
    """
    阶梯分成函数
    收益越高，分成比例越高
    """
    def __init__(self, tiers: List[Dict]):
        """
        tiers: [
            {"threshold": 100000, "ratio": 0.1},
            {"threshold": 500000, "ratio": 0.2},
            {"threshold": float('inf'), "ratio": 0.3}
        ]
        """
        self.tiers = sorted(tiers, key=lambda t: t["threshold"])

    def apply(self, base_amount: float) -> float:
        total_share = 0
        prev_threshold = 0

        for tier in self.tiers:
            if base_amount > prev_threshold:
                tier_amount = min(base_amount, tier["threshold"]) - prev_threshold
                total_share += tier_amount * tier["ratio"]
                prev_threshold = tier["threshold"]

        return total_share


class GuaranteeFunction:
    """
    保底函数基类
    确保最低收益或最高成本
    """
    def apply(self, base_amount: float) -> float:
        raise NotImplementedError


class MinimumGuaranteeFunction(GuaranteeFunction):
    """最低收益保底函数"""
    def __init__(self, min_amount: float):
        self.min_amount = min_amount

    def apply(self, base_amount: float) -> float:
        return max(self.min_amount, base_amount)


class MaximumCostFunction(GuaranteeFunction):
    """最高成本封顶函数"""
    def __init__(self, max_amount: float):
        self.max_amount = max_amount

    def apply(self, base_amount: float) -> float:
        return min(self.max_amount, base_amount)


class PenaltyFunction:
    """
    违约函数基类
    定义违约条件和罚款计算
    """
    def __init__(self, penalty_rate: float, tolerance_pct: float = 0.05):
        self.penalty_rate = penalty_rate  # 罚款率（元/kWh偏差）
        self.tolerance = tolerance_pct  # 免考核偏差范围

    def calculate_penalty(self, contracted_energy: float,
                          actual_energy: float, price: float) -> float:
        """
        计算违约罚款
        """
        deviation = abs(actual_energy - contracted_energy) / contracted_energy

        if deviation <= self.tolerance:
            return 0  # 免考核

        excess_deviation = deviation - self.tolerance
        penalty = excess_deviation * contracted_energy * self.penalty_rate
        return penalty
```

```python
class ContractTemplateManager:
    """
    合同模板管理器
    管理多版本合同模板，支持快速生成合同实例
    """
    def __init__(self):
        self.templates = {}  # {template_id: ContractTemplate}

    def register_template(self, template_id: str, template: 'ContractTemplate'):
        """注册合同模板"""
        self.templates[template_id] = template

    def instantiate(self, template_id: str, params: Dict) -> ContractMeta:
        """
        从模板实例化合同
        """
        template = self.templates.get(template_id)
        if not template:
            raise ValueError(f"Template {template_id} not found")

        return template.generate(params)

    def list_templates(self) -> List[Dict]:
        """列出所有可用模板"""
        return [
            {
                "template_id": tid,
                "name": t.name,
                "description": t.description,
                "version": t.version
            }
            for tid, t in self.templates.items()
        ]


class ContractTemplate:
    """
    合同模板
    预定义的合同条款模板
    """
    def __init__(self, template_id: str, name: str,
                 description: str, version: int = 1):
        self.template_id = template_id
        self.name = name
        self.description = description
        self.version = version
        self.price_function_builder = None
        self.share_function_builder = None
        self.guarantee_function_builder = None

    def generate(self, params: Dict) -> ContractMeta:
        """
        根据参数生成合同实例
        """
        contract = ContractMeta(
            contract_id=f"{self.template_id}_{datetime.now().strftime('%Y%m%d%H%M%S')}",
            contract_type=params.get("contract_type", ContractType.RETAIL_FIXED),
            start_date=params["start_date"],
            end_date=params["end_date"],
            counterparty=params["counterparty"],
            capacity_kw=params["capacity_kw"]
        )

        # 添加电价函数
        if self.price_function_builder:
            price_func = self.price_function_builder(params)
            contract.add_price_function(price_func)

        # 添加分成函数
        if self.share_function_builder:
            share_func = self.share_function_builder(params)
            contract.add_share_function(share_func)

        # 添加保底函数
        if self.guarantee_function_builder:
            guarantee_func = self.guarantee_function_builder(params)
            contract.add_guarantee_function(guarantee_func)

        return contract
```

## 2. 动态电价响应

### 2.1 动态电价模式

HEMS/VPP系统支持四种动态电价模式：

| 电价模式              | 说明                            | 更新频率      | 响应策略                |
| --------------------- | ------------------------------- | ------------- | ----------------------- |
| **分时电价（TOU）**   | 固定峰/谷/平三段式电价          | 每日/季度更新 | 日前优化 + 日内执行     |
| **实时电价（RTP）**   | 每5-30分钟更新的现货市场价格    | 5-30分钟      | 实时跟踪 + 预测修正     |
| **尖峰电价（CPP）**   | 极端天气/电网紧张时触发的高电价 | 事件触发      | 预冷/预热 + 储能放电    |
| **需求侧竞价（DSB）** | 用户主动报价削减负荷            | 事件触发      | 用户设定底价 + 自动竞价 |

### 2.2 动态电价数据获取

```python
class DynamicPriceClient:
    """
    动态电价数据获取客户端
    支持多种数据源接入
    """
    def __init__(self):
        self.price_sources = {}
        self.current_price = 0.0
        self.price_curve = []  # 未来24小时电价曲线
        self.last_update = 0

    def register_source(self, name, source_type, config):
        """
        注册电价数据源
        source_type: "RETAILER_API" | "AEMO" | "VPP_PLATFORM" | "MANUAL"
        """
        self.price_sources[name] = {
            "type": source_type,
            "config": config,
            "last_success": 0
        }

    def fetch_current_price(self):
        """
        获取当前实时电价
        支持多种数据源自动切换
        """
        for name, source in self.price_sources.items():
            try:
                if source["type"] == "RETAILER_API":
                    price = self._fetch_retailer_api(source["config"])
                elif source["type"] == "AEMO":
                    price = self._fetch_aemo_price(source["config"])
                elif source["type"] == "VPP_PLATFORM":
                    price = self._fetch_vpp_price(source["config"])
                else:
                    continue

                self.current_price = price
                self.last_update = time.time()
                source["last_success"] = time.time()
                return price
            except Exception as e:
                print(f"Source {name} failed: {e}")
                continue

        # 所有数据源失败，使用最后一次有效价格
        return self.current_price

    def fetch_price_forecast(self, hours_ahead=24):
        """
        获取未来电价预测曲线
        用于日前优化和日内滚动修正
        """
```

### 2.3 动态电价响应算法

```python
class DynamicPriceResponder:
    """
    动态电价响应引擎
    根据实时电价和预测电价，动态调整家庭能源调度策略
    """
    def __init__(self, price_client, battery, ev, hvac, heat_pump):
        self.price = price_client
        self.battery = battery
        self.ev = ev
        self.hvac = hvac
        self.heat_pump = heat_pump

        # 可配置参数
        self.price_thresholds = {
            "buy_low": 0.25,    # 低价买入阈值 (元/kWh)
            "sell_high": 0.60,  # 高价卖出阈值 (元/kWh)
            "critical": 1.00,   # 尖峰电价阈值 (元/kWh)
        }

        # 用户偏好
        self.user_preferences = {
            "min_comfort_temp": 24,
            "max_comfort_temp": 28,
            "min_hot_water_temp": 40,
            "ev_min_soc": 60,
        }

    def respond_to_current_price(self, current_price, pv_power, load_power):
        """
        对当前电价做出实时响应（1秒周期执行）
        """
        # 1. 判断当前电价区间
        price_level = self._classify_price(current_price)

        # 2. 根据电价区间执行对应策略
        if price_level == "CRITICAL":
            return self._critical_price_response(current_price)
        elif price_level == "HIGH":
            return self._high_price_response(current_price)
        elif price_level == "LOW":
            return self._low_price_response(current_price, pv_power)
        else:  # NORMAL
            return self._normal_price_response(pv_power, load_power)

    def _classify_price(self, price):
        """电价分类"""
        if price >= self.price_thresholds["critical"]:
            return "CRITICAL"
        elif price >= self.price_thresholds["sell_high"]:
            return "HIGH"
        elif price <= self.price_thresholds["buy_low"]:
            return "LOW"
        else:
            return "NORMAL"

    def _critical_price_response(self, price):
        """
        尖峰电价响应（极端高价）
        目标：最大化放电，最小化购电
        """
        actions = []

        # 1. 储能全力放电
        if self.battery.soc > self.battery.min_soc:
            discharge = self.battery.max_discharge
            actions.append(("battery", "discharge", discharge))

        # 2. EV V2G放电
        if self.ev.is_connected and self.ev.soc > self.user_preferences["ev_min_soc"] + 20:
            v2g_power = min(
                self.ev.max_discharge,
                (self.ev.soc - self.user_preferences["ev_min_soc"]) / 100 * self.ev.capacity * 1000
            )
            actions.append(("ev_v2g", "discharge", v2g_power))

        # 3. 空调温度偏移
        actions.append(("hvac", "temp_offset", 3.0))

        # 4. 热泵停止加热
        actions.append(("heat_pump", "stop", 0))

        # 5. 切断所有L4设备
        actions.append(("pool", "stop", 0))
        actions.append(("appliances", "delay_all", 0))

        return actions

    def _high_price_response(self, price):
        """
        高电价响应
        目标：减少购电，利用储能放电
        """
        actions = []

        # 1. 储能放电
        if self.battery.soc > self.battery.min_soc + 10:
            discharge = min(self.battery.max_discharge * 0.7, self.battery.max_discharge)
            actions.append(("battery", "discharge", discharge))

        # 2. 空调温度偏移
        actions.append(("hvac", "temp_offset", 1.0))

        # 3. 热泵延时加热
        if self.heat_pump.can_delay(hours=2):
            actions.append(("heat_pump", "delay", 2))

        # 4. 限制EV充电功率
        if self.ev.is_charging:
            actions.append(("ev", "limit_power", self.ev.rated_power * 0.5))

        return actions

    def _low_price_response(self, price, pv_power):
        """
        低电价响应
        目标：最大化充电，预冷/预热
        """
        actions = []

        # 1. 储能全力充电
        if self.battery.soc < self.battery.max_soc:
            charge = self.battery.max_charge
            actions.append(("battery", "charge", charge))

        # 2. EV充电
        if self.ev.is_connected and self.ev.soc < 100:
            actions.append(("ev", "charge", self.ev.max_charge))

        # 3. 空调预冷/预热
        actions.append(("hvac", "pre_condition", 0))

        # 4. 热泵加热
        if self.heat_pump.water_temp < self.heat_pump.max_temp:
            actions.append(("heat_pump", "heat", self.heat_pump.rated_power))

        # 5. 启动泳池泵和加热器
        actions.append(("pool", "start", 0))

        # 6. 启动延时家电
        actions.append(("appliances", "start_available", 0))

        return actions

    def _normal_price_response(self, pv_power, load_power):
        """
        正常电价响应
        按自发自用模式运行
        """
        return self._self_consumption_mode(pv_power, load_power)

    def _self_consumption_mode(self, pv_power, load_power):
        """自发自用模式"""
        actions = []
        net_power = pv_power - load_power

        if net_power > 0:  # 光伏有余
            if self.battery.soc < self.battery.max_soc:
                actions.append(("battery", "charge", min(net_power, self.battery.max_charge)))
        else:  # 光伏不足
            if self.battery.soc > self.battery.min_soc:
                actions.append(("battery", "discharge", min(abs(net_power), self.battery.max_discharge)))

        return actions
```

```python
def optimize_with_dynamic_price(pv_forecast, load_forecast, price_forecast,
                                battery_params, ev_params, hvac_params):
    """
    基于动态电价的日前优化调度

def analyze_price_signal(price_forecast):
    """
    分析电价信号，识别关键时段
    """
    import numpy as np

    prices = [p["price"] for p in price_forecast]
    mean_price = np.mean(prices)
    std_price = np.std(prices)

    # 识别尖峰电价时段（超过均值+2倍标准差）
    critical_threshold = mean_price + 2 * std_price
    critical_periods = [
        p for p in price_forecast
        if p["price"] >= critical_threshold
    ]

    # 识别低价时段（低于均值-1倍标准差）
    low_threshold = mean_price - std_price
    low_price_periods = [
        p for p in price_forecast
        if p["price"] <= low_threshold
    ]

    # 识别高价时段（高于均值+1倍标准差，但未到尖峰）
    high_threshold = mean_price + std_price
    high_price_periods = [
        p for p in price_forecast
        if high_threshold <= p["price"] < critical_threshold
    ]

    return {
        "critical_periods": critical_periods,
        "low_price_periods": low_price_periods,
        "high_price_periods": high_price_periods,
        "mean_price": mean_price,
        "std_price": std_price
    }


def optimize_battery_with_price(price_forecast, battery_params,
                                 charge_periods, discharge_periods):
    """基于电价的储能优化调度"""
    schedule = []
    for period in price_forecast:
        if period in charge_periods:
            schedule.append({"action": "charge", "power": battery_params["max_charge"]})
        elif period in discharge_periods:
            schedule.append({"action": "discharge", "power": battery_params["max_discharge"]})
        else:
            schedule.append({"action": "idle", "power": 0})
    return schedule


def optimize_ev_charging(price_forecast, ev_params, preferred_periods):
    """基于电价的EV充电优化"""
    schedule = []
    for period in price_forecast:
        if period in preferred_periods:
            schedule.append({"action": "charge", "power": ev_params["max_charge"]})
        else:
            schedule.append({"action": "idle", "power": 0})
    return schedule


def optimize_hvac_with_price(price_forecast, hvac_params,
                              pre_cool_periods, shed_periods):
    """基于电价的空调优化调度"""
    schedule = []
    for period in price_forecast:
        if period in pre_cool_periods:
            schedule.append({"action": "pre_cool", "temp_offset": -2})
        elif period in shed_periods:

def optimize_heat_pump_with_price(price_forecast, pv_forecast, preferred_periods):

            schedule.append({"action": "heat", "power": 3.0})
```

### 3.3 M&V测量与验证

基于IPMVP（国际节能效果测量与验证协议）的M&V框架：

```python
class MeasurementAndVerification:
    """
    M&V测量与验证引擎
    基于IPMVP协议计算节能效益
    """
    def __init__(self, baseline_data, adjustment_factors=None):
        self.baseline = baseline_data  # 基准期数据
        self.adjustment = adjustment_factors or {}  # 调整因子
        self.reporting_period_data = []

    def set_baseline(self, data):
        """
        设置基准期数据
        通常取改造前12个月的数据
        """
        self.baseline = data

    def add_reporting_data(self, data_point):
        """
        添加报告期数据
        """
        self.reporting_period_data.append(data_point)

    def calculate_energy_savings(self, method="OPTION_C"):
        """
        计算节能量
        method: IPMVP选项
        - OPTION_A: 部分测量+估算
        - OPTION_B: 全部测量
        - OPTION_C: 整体设施测量
        - OPTION_D: 校准模拟
        """
        if method == "OPTION_C":
            return self._calculate_option_c()
        elif method == "OPTION_D":
            return self._calculate_option_d()
        else:
            return self._calculate_option_ab()

    def _calculate_option_c(self):
        """
        OPTION C：整体设施测量
        节能量 = (基准期能耗 - 报告期能耗) ± 调整量
        """
        baseline_energy = sum(self.baseline)
        reporting_energy = sum(
            d["energy_kwh"] for d in self.reporting_period_data
        )

        # 调整量（天气、产量等因素）
        adjustment = self._calculate_adjustment()

        savings = baseline_energy - reporting_energy + adjustment

        return {
            "method": "OPTION_C",
            "baseline_energy_kwh": baseline_energy,
            "reporting_energy_kwh": reporting_energy,
            "adjustment_kwh": adjustment,
            "net_savings_kwh": savings,
            "savings_pct": (savings / baseline_energy * 100) if baseline_energy > 0 else 0
        }

    def _calculate_option_d(self):
        """
        OPTION D：校准模拟
        使用模拟模型计算节能量
        """
        # 实际中调用建筑能耗模拟引擎
        simulated_baseline = sum(self.baseline) * 0.95  # 简化
        actual_energy = sum(
            d["energy_kwh"] for d in self.reporting_period_data
        )

        savings = simulated_baseline - actual_energy

        return {
            "method": "OPTION_D",
            "simulated_baseline_kwh": simulated_baseline,
            "actual_energy_kwh": actual_energy,
            "net_savings_kwh": savings,
            "savings_pct": (savings / simulated_baseline * 100) if simulated_baseline > 0 else 0
        }

    def _calculate_option_ab(self):
        """OPTION A/B：部分或全部测量"""
        # 简化实现
        return {"method": "OPTION_AB", "net_savings_kwh": 0}

    def _calculate_adjustment(self):
        """
        计算调整量
        考虑天气、产量、使用时间等因素
        """
        adjustment = 0

        # 天气调整
        if "temperature" in self.adjustment:
            temp_diff = self.adjustment["temperature"]
            adjustment += temp_diff * 100  # 每度温差影响100kWh

        # 产量调整
        if "production" in self.adjustment:
            prod_change = self.adjustment["production"]
            adjustment += prod_change * 0.5  # 每单位产量变化影响0.5kWh

        return adjustment

    def verify_savings(self, guaranteed_savings_kwh, tolerance_pct=0.1):
        """
        验证节电保证
        判断是否达到承诺的节电效果
        """
        result = self.calculate_energy_savings()
        actual_savings = result["net_savings_kwh"]

        # 计算偏差
        deviation = (actual_savings - guaranteed_savings_kwh) / guaranteed_savings_kwh

        is_achieved = deviation >= -tolerance_pct

        return {
            "guaranteed_savings_kwh": guaranteed_savings_kwh,
            "actual_savings_kwh": actual_savings,
            "deviation_pct": deviation * 100,
            "is_achieved": is_achieved,
            "shortfall_kwh": max(0, guaranteed_savings_kwh - actual_savings) if not is_achieved else 0
        }
```

```python
class EMCSettlementEngine:
    """
    EMC结算引擎
    计算节能收益并按合同约定分配
    """
    def __init__(self, contract):
        self.contract = contract  # EMC合同元数据
        self.mv = MeasurementAndVerification(baseline_data=[])

    def calculate_period_settlement(self, period_start, period_end):
        """
        计算结算周期内的收益分配
        """
        # 1. 计算节能量
        savings_result = self.mv.calculate_energy_savings()

        # 2. 计算节能收益（按当地电价）
        energy_price = self._get_average_price(period_start, period_end)
        total_savings_value = savings_result["net_savings_kwh"] * energy_price

        # 3. 按合同分成比例分配
        vpp_share_ratio = self._get_vpp_share_ratio(total_savings_value)
        client_share_ratio = 1 - vpp_share_ratio
```

```python
class CarbonAccountingEngine:
    """
    碳排放核算引擎
    支持范围1/2/3碳排放计算
    """
    def __init__(self):
        # 电网排放因子（kg CO2/kWh）
        self.grid_emission_factor = 0.581  # 2025年全国平均
        # 各省排放因子
        self.province_factors = {
            "四川": 0.152,  # 水电为主，排放因子低
            "甘肃": 0.412,
            "江苏": 0.682,
            "广东": 0.512
        }

    def calculate_scope2_emissions(self, grid_consumption_kwh, province="四川"):
        """
        计算范围2排放（外购电力）
        """
        factor = self.province_factors.get(province, self.grid_emission_factor)
        emissions = grid_consumption_kwh * factor / 1000  # 吨CO2
        return emissions

    def calculate_avoided_emissions(self, pv_generation_kwh, province="四川"):
        """
        计算光伏发电避免的碳排放
        """
        factor = self.province_factors.get(province, self.grid_emission_factor)
        avoided = pv_generation_kwh * factor / 1000  # 吨CO2
        return avoided

    def calculate_green_certificate_value(self, green_energy_kwh):
        """
        计算绿电消费的碳减排价值
        """
        # 1个绿证 = 1000 kWh绿电
        certificates = green_energy_kwh / 1000
        # 碳减排量
        emission_reduction = green_energy_kwh * self.grid_emission_factor / 1000
        return {
            "green_certificates": certificates,
            "emission_reduction_tco2": emission_reduction,
            "carbon_value": emission_reduction * 80  # 碳价80元/吨
        }

    def generate_carbon_report(self, period_start, period_end, energy_data):
        """
        生成碳排放报告
        适用于Apple产业链企业的碳报告要求
        """
        total_emissions = 0
        total_avoided = 0

        for data in energy_data:
            if data["type"] == "grid_import":
                total_emissions += self.calculate_scope2_emissions(
                    data["energy_kwh"], data.get("province", "四川")
                )
            elif data["type"] == "pv_generation":
                total_avoided += self.calculate_avoided_emissions(
                    data["energy_kwh"], data.get("province", "四川")
                )

        return {
            "report_period": f"{period_start}/{period_end}",
            "total_scope2_emissions_tco2": total_emissions,
            "total_avoided_emissions_tco2": total_avoided,
            "net_emissions_tco2": max(0, total_emissions - total_avoided),
            "renewable_energy_pct": self._calculate_renewable_pct(energy_data),
            "certified_by": "VPP_Carbon_Accounting_Engine_v1.0"
        }

    def _calculate_renewable_pct(self, energy_data):
        """计算可再生能源占比"""
        total = sum(d["energy_kwh"] for d in energy_data)
        renewable = sum(
            d["energy_kwh"] for d in energy_data
            if d["type"] in ["pv_generation", "green_purchase"]
        )
        return (renewable / total * 100) if total > 0 else 0
```

```python
class CBAMComplianceEngine:
    """
    CBAM（欧盟碳边境调节机制）合规引擎
    帮助出口企业满足欧盟碳关税要求
    """
    def __init__(self):
        # CBAM默认排放因子（tCO2/吨产品）
        self.default_factors = {
            "钢铁": 1.85,
            "铝": 1.65,
            "水泥": 0.65,
            "化肥": 2.10,
            "电力": 0.581
        }
        # 欧盟碳价（欧元/吨）
        self.eu_carbon_price = 80  # 2025年预估

    def calculate_cbam_certificate_cost(self, product_type, quantity_tons,
                                         actual_emissions_tco2=None):
        """
        计算CBAM证书成本
        """
        # 默认排放量
        default_factor = self.default_factors.get(product_type, 1.0)
        default_emissions = quantity_tons * default_factor

        # 实际排放量（如果提供）
        if actual_emissions_tco2:
            actual_emissions = actual_emissions_tco2
        else:
            actual_emissions = default_emissions

        # 计算CBAM证书数量
        # CBAM证书 = (默认排放 - 实际排放) * 碳价
        # 如果实际排放低于默认值，可以减少CBAM证书购买
        certificate_quantity = max(0, default_emissions - actual_emissions)

        # 证书成本
        certificate_cost = certificate_quantity * self.eu_carbon_price

        # 绿电抵扣
        green_electricity_deduction = self._calculate_green_electricity_deduction(
            product_type, quantity_tons
        )

        return {
            "product_type": product_type,
            "quantity_tons": quantity_tons,
            "default_emissions_tco2": default_emissions,
            "actual_emissions_tco2": actual_emissions,
            "certificate_quantity": certificate_quantity,
            "certificate_cost_eur": certificate_cost,
            "green_electricity_deduction_tco2": green_electricity_deduction,
            "net_certificate_cost_eur": max(0, certificate_cost - green_electricity_deduction * self.eu_carbon_price)
        }

    def _calculate_green_electricity_deduction(self, product_type, quantity_tons):
        """
        计算绿电消费的CBAM抵扣
        """
        # 每吨产品的电力消耗（MWh）
        electricity_intensity = {
            "钢铁": 0.5,  # MWh/吨
            "铝": 15.0,   # 电解铝耗电高
            "水泥": 0.1,
            "化肥": 0.3
        }

        intensity = electricity_intensity.get(product_type, 0.1)
        total_electricity_mwh = quantity_tons * intensity

        # 假设绿电占比
        green_pct = 0.3  # 30%绿电
        green_electricity_mwh = total_electricity_mwh * green_pct

        # 绿电减排量
        deduction = green_electricity_mwh * self.default_factors["电力"]
        return deduction
```

```python
        """
        交易绿证
        """
        cert = next((c for c in self.certificates if c["certificate_id"] == cert_id), None)
        if not cert or cert["status"] != "AVAILABLE":
            return None

        trade = {
            "trade_id": f"TRADE-{int(time.time())}",
            "certificate_id": cert_id,
            "seller": "VPP_Platform",
            "buyer": buyer,
            "price": price_per_cert,
            "total_amount": price_per_cert,
            "timestamp": int(time.time() * 1000),
            "status": "COMPLETED"
        }

        cert["status"] = "SOLD"
        cert["buyer"] = buyer
        self.trades.append(trade)

        return trade

    def retire_certificate(self, cert_id, purpose):
        """

            "retired_for": purpose,

            "energy_mwh": cert["energy_mwh"],
            "source": cert["source"]
```

```python
class RevenueCalculator:
    """

    实时计算理论收益与实际收益，分析偏差

            self_use_saving = pv_self_use * price

            # 光伏余电上网
```
