### 2.2 OCPP 1.6J核心实现

```python
import asyncio
import websockets
import json
from datetime import datetime
from enum import Enum
from typing import Optional, Callable, Dict, List


class OCPPVersion(Enum):
    """OCPP版本枚举"""
    V16 = "ocpp1.6"
    V201 = "ocpp2.0.1"


class ChargePointStatus(Enum):
    """充电桩状态枚举"""
    AVAILABLE = "Available"
    PREPARING = "Preparing"
    CHARGING = "Charging"
    SUSPENDED_EV = "SuspendedEV"
    SUSPENDED_EVSE = "SuspendedEVSE"
    FINISHING = "Finishing"
    RESERVED = "Reserved"
    UNAVAILABLE = "Unavailable"
    FAULTED = "Faulted"


class OCPPClient:
    """
    OCPP 1.6J 客户端实现
    运行于充电桩端，与中央系统通信
    """
    def __init__(self, charge_point_id: str, central_system_url: str):
        self.charge_point_id = charge_point_id
        self.central_system_url = central_system_url
        self.websocket: Optional[websockets.WebSocketClientProtocol] = None
        self.status = ChargePointStatus.UNAVAILABLE
        self.transaction_id: Optional[int] = None
        self.callbacks: Dict[str, Callable] = {}

        # 连接状态
        self.is_connected = False
        self.reconnect_attempts = 0
        self.max_reconnect_attempts = 10

    async def connect(self):
        """
        连接到中央系统
        使用WSS协议，支持双向认证
        """
        uri = f"{self.central_system_url}/{self.charge_point_id}"
        try:
            self.websocket = await websockets.connect(
                uri,
                subprotocols=["ocpp1.6"],
                ping_interval=30,
                ping_timeout=10
            )
            self.is_connected = True
            self.reconnect_attempts = 0

            # 发送BootNotification
            await self.send_boot_notification()

            # 启动消息接收循环
            asyncio.create_task(self._receive_loop())

        except Exception as e:
            print(f"Connection failed: {e}")
            await self._reconnect()

    async def disconnect(self):
        """断开连接"""
        if self.websocket:
            await self.websocket.close()
        self.is_connected = False

    async def send_boot_notification(self):
        """
        发送BootNotification
        充电桩启动时通知中央系统
        """
        request = {
            "chargePointVendor": "VPP_EVSE",
            "chargePointModel": "VPP-AC-7kW-v3",
            "chargePointSerialNumber": self.charge_point_id,
            "firmwareVersion": "3.2.1",
            "iccid": "89860123456789012345",
            "imsi": "460011234567890"
        }

        response = await self._call("BootNotification", request)
        if response:
            self.status = ChargePointStatus.AVAILABLE
            print(f"BootNotification accepted: {response}")

    async def send_heartbeat(self):
        """
        发送心跳
        每30秒发送一次
        """
        while self.is_connected:
            await asyncio.sleep(30)
            try:
                response = await self._call("Heartbeat", {})
                if response:
                    print(f"Heartbeat OK: {response}")
            except Exception as e:
                print(f"Heartbeat failed: {e}")
                await self._reconnect()

    async def start_transaction(self, id_tag: str, meter_start: int,
                                 reservation_id: Optional[int] = None):
        """
        启动充电交易
        """
        request = {
            "idTag": id_tag,
            "meterStart": meter_start,
            "timestamp": datetime.utcnow().isoformat(),
            "reservationId": reservation_id
        }

        response = await self._call("StartTransaction", request)
        if response and response.get("status") == "Accepted":
            self.transaction_id = response["transactionId"]
            self.status = ChargePointStatus.CHARGING
            return response
        return None

    async def stop_transaction(self, transaction_id: int, meter_stop: int,
                                id_tag: Optional[str] = None):
        """
        停止充电交易
        """
        request = {
            "transactionId": transaction_id,
            "meterStop": meter_stop,
            "timestamp": datetime.utcnow().isoformat(),
            "idTag": id_tag
        }

        response = await self._call("StopTransaction", request)
        if response:
            self.transaction_id = None
            self.status = ChargePointStatus.AVAILABLE
            return response
        return None

    async def send_meter_values(self, power_w: float, energy_kwh: float,
                                  soc: Optional[float] = None):
        """
        上报电表读数
        每秒上报一次功率，每5分钟上报一次累计电量
        """
        sampled_value = [
            {
                "measurand": "Power.Active.Import",
                "unit": "W",
                "value": power_w
            },
            {
                "measurand": "Energy.Active.Import.Register",
                "unit": "kWh",
                "value": energy_kwh
            }
        ]

        if soc is not None:
            sampled_value.append({
                "measurand": "SoC",
                "unit": "%",
                "value": soc
            })

        request = {
            "transactionId": self.transaction_id,
            "meterValue": [{
                "timestamp": datetime.utcnow().isoformat(),
                "sampledValue": sampled_value
            }]
        }

        await self._call("MeterValues", request)

    async def set_charging_profile(self, limit_w: int, duration_s: int):
        """
        设置充电曲线
        用于动态负载平衡和V2G调度
        """
        request = {
            "connectorId": 1,
            "csChargingProfiles": {
                "chargingProfileId": int(datetime.now().timestamp()),
                "stackLevel": 1,
                "chargingProfilePurpose": "TxProfile",
                "chargingProfileKind": "Relative",
                "chargingSchedule": {
                    "duration": duration_s,
                    "chargingRateUnit": "W",
                    "chargingSchedulePeriod": [{
                        "startPeriod": 0,
                        "limit": limit_w
                    }]
                }
            }
        }

        response = await self._call("SetChargingProfile", request)
        return response

    async def remote_start_transaction(self, id_tag: str):
        """
        远程启动充电
        """
        request = {
            "idTag": id_tag,
            "connectorId": 1
        }
        response = await self._call("RemoteStartTransaction", request)
        return response

    async def remote_stop_transaction(self, transaction_id: int):
        """
        远程停止充电
        """
        request = {"transactionId": transaction_id}
        response = await self._call("RemoteStopTransaction", request)
        return response

    async def _call(self, action: str, payload: dict) -> Optional[dict]:
        """
        发送OCPP调用请求
        使用JSON-RPC over WebSocket
        """
        if not self.websocket:
            return None

        message_id = f"{self.charge_point_id}-{int(datetime.now().timestamp()*1000)}"
        call = [2, message_id, action, payload]

        try:
            await self.websocket.send(json.dumps(call))
            # 等待响应（简化实现）
            response = await asyncio.wait_for(
                self.websocket.recv(), timeout=30
            )
            return self._parse_response(json.loads(response))
        except Exception as e:
            print(f"Call failed: {action} - {e}")
            return None

    def _parse_response(self, message: list) -> Optional[dict]:
        """解析OCPP响应"""
        if len(message) >= 3 and message[0] == 3:
            return message[2]
        return None

    async def _receive_loop(self):
        """消息接收循环"""
        while self.is_connected:
            try:
                message = await self.websocket.recv()
                data = json.loads(message)

                # 处理远程调用（来自中央系统的指令）
                if data[0] == 2:  # Call
                    action = data[2]
                    payload = data[3]
                    await self._handle_remote_call(data[1], action, payload)

            except websockets.exceptions.ConnectionClosed:
                print("Connection closed")
                await self._reconnect()
                break
            except Exception as e:
                print(f"Receive error: {e}")

    async def _handle_remote_call(self, message_id: str, action: str,
                                    payload: dict):
        """处理远程调用"""
        if action == "RemoteStartTransaction":
            result = await self.remote_start_transaction(payload.get("idTag"))
            await self._send_response(message_id, result)
        elif action == "RemoteStopTransaction":
            result = await self.remote_stop_transaction(payload.get("transactionId"))
            await self._send_response(message_id, result)
        elif action == "SetChargingProfile":
            result = await self.set_charging_profile(
                payload.get("limit_w", 7000),
                payload.get("duration_s", 3600)
            )
            await self._send_response(message_id, result)
        else:
            await self._send_response(message_id, {"status": "NotImplemented"})

    async def _send_response(self, message_id: str, result: dict):
        """发送响应"""
        response = [3, message_id, result]
        await self.websocket.send(json.dumps(response))

    async def _reconnect(self):
        """重连机制（指数退避）"""
        while self.reconnect_attempts < self.max_reconnect_attempts:
            wait_time = min(30, 2 ** self.reconnect_attempts)
            print(f"Reconnecting in {wait_time}s (attempt {self.reconnect_attempts + 1})")
            await asyncio.sleep(wait_time)

            try:
                await self.connect()
                if self.is_connected:
                    print("Reconnected successfully")
                    return
            except:
                self.reconnect_attempts += 1

        print("Max reconnect attempts reached")
        self.status = ChargePointStatus.FAULTED
```

```python
class OCPPCentralSystem:
    """
    OCPP中央系统
    管理所有充电桩的连接和通信
    """
    def __init__(self, host: str = "0.0.0.0", port: int = 8080):
        self.host = host
        self.port = port
        self.charge_points: Dict[str, OCPPClient] = {}
        self.transactions: Dict[int, dict] = {}

    async def start(self):
        """启动WebSocket服务器"""
        async with websockets.serve(
            self._handle_connection,
            self.host,
            self.port,
            subprotocols=["ocpp1.6", "ocpp2.0.1"]
        ):
            print(f"OCPP Central System started on {self.host}:{self.port}")
            await asyncio.Future()  # 永久运行

    async def _handle_connection(self, websocket, path):
        """
        处理充电桩连接
        path格式: /{charge_point_id}
        """
        charge_point_id = path.strip("/")
        print(f"New connection from {charge_point_id}")

        # 创建充电桩实例
        cp = OCPPClient(charge_point_id, "")
        cp.websocket = websocket
        self.charge_points[charge_point_id] = cp

        try:
            async for message in websocket:
                await self._handle_message(charge_point_id, message)
        except websockets.exceptions.ConnectionClosed:
            print(f"Connection closed: {charge_point_id}")
        finally:
            if charge_point_id in self.charge_points:
                del self.charge_points[charge_point_id]

    async def _handle_message(self, charge_point_id: str, message: str):
        """处理接收到的消息"""
        data = json.loads(message)
        if data[0] == 2:  # Call
            message_id = data[1]
            action = data[2]
            payload = data[3]

            # 处理不同动作
            if action == "BootNotification":
                response = await self._handle_boot_notification(charge_point_id, payload)
            elif action == "Heartbeat":
                response = {"currentTime": datetime.utcnow().isoformat()}
            elif action == "StartTransaction":
                response = await self._handle_start_transaction(charge_point_id, payload)
            elif action == "StopTransaction":
                response = await self._handle_stop_transaction(charge_point_id, payload)
            elif action == "MeterValues":
                response = await self._handle_meter_values(charge_point_id, payload)
            else:
                response = {"status": "NotImplemented"}

            # 发送响应
            cp = self.charge_points.get(charge_point_id)
            if cp and cp.websocket:
                await cp.websocket.send(json.dumps([3, message_id, response]))

    async def _handle_boot_notification(self, charge_point_id: str,
                                          payload: dict) -> dict:
        """处理BootNotification"""
        print(f"BootNotification from {charge_point_id}: {payload}")
        return {
            "status": "Accepted",
            "currentTime": datetime.utcnow().isoformat(),
            "interval": 30  # 心跳间隔
        }

    async def _handle_start_transaction(self, charge_point_id: str,
                                          payload: dict) -> dict:
        """处理StartTransaction"""
        transaction_id = int(datetime.now().timestamp() * 1000)
        self.transactions[transaction_id] = {
            "charge_point_id": charge_point_id,
            "id_tag": payload.get("idTag"),
            "meter_start": payload.get("meterStart"),
            "timestamp": payload.get("timestamp"),
            "status": "ACTIVE"
        }
        return {
            "transactionId": transaction_id,
            "idTagInfo": {"status": "Accepted"}
        }

    async def _handle_stop_transaction(self, charge_point_id: str,
                                         payload: dict) -> dict:
        """处理StopTransaction"""
        transaction_id = payload.get("transactionId")
        if transaction_id in self.transactions:
            self.transactions[transaction_id]["status"] = "COMPLETED"
            self.transactions[transaction_id]["meter_stop"] = payload.get("meterStop")
        return {"idTagInfo": {"status": "Accepted"}}

    async def _handle_meter_values(self, charge_point_id: str,
                                     payload: dict) -> dict:
        """处理MeterValues"""
        # 存储电表读数
        return {}

    async def broadcast_set_charging_profile(self, limit_w: int,
                                              charge_point_ids: List[str] = None):
        """
        广播充电曲线设置
        用于动态负载平衡

        targets = charge_point_ids or list(self.charge_points.keys())
        tasks = []
        for cp_id in targets:
            cp = self.charge_points.get(cp_id)
            if cp:
                tasks.append(cp.set_charging_profile(limit_w, 3600))
        await asyncio.gather(*tasks)
```

```python
class DynamicLoadBalancer:
    """
    动态负载平衡算法
    在台区容量限制下，最优分配充电功率
    """
    def __init__(self, transformer_capacity_va: float,
                 safety_margin_pct: float = 0.1):
        self.transformer_capacity = transformer_capacity_va
        self.safety_margin = safety_margin_pct
        self.chargers: Dict[str, ChargerState] = {}

    def register_charger(self, charger_id: str, max_power_w: float,
                          priority: int = 3):
        """
        注册充电桩
        priority: 1=VIP, 2=常规, 3=机会
        """
        self.chargers[charger_id] = ChargerState(
            charger_id=charger_id,
            max_power=max_power_w,
            priority=priority
        )

    def calculate_available_capacity(self, current_load_va: float) -> float:
        """
        计算可用充电容量
        """
        max_capacity = self.transformer_capacity * (1 - self.safety_margin)
        available = max(0, max_capacity - current_load_va)
        return available

    def allocate_power(self, current_load_va: float) -> Dict[str, float]:
        """
        分配充电功率
        返回 {charger_id: allocated_power_w}
        """
        available_w = self.calculate_available_capacity(current_load_va) * 1000  # VA→W近似
        allocation = {}

        # 按优先级分组
        priority_groups = {1: [], 2: [], 3: []}
        for charger_id, state in self.chargers.items():
            if state.is_charging:
                priority_groups[state.priority].append(charger_id)

        # 按优先级分配
        for priority in [1, 2, 3]:
            group = priority_groups[priority]
            if not group or available_w <= 0:
                # 低优先级桩分配0功率
                for charger_id in group:
                    allocation[charger_id] = 0
                continue

            # 计算该组总需求
            total_demand = sum(
                self.chargers[cid].requested_power
                for cid in group
            )

            if total_demand <= available_w:
                # 需求小于可用容量，全额分配
                for charger_id in group:
                    allocation[charger_id] = self.chargers[charger_id].requested_power
                available_w -= total_demand
            else:
                # 需求大于可用容量，按比例分配
                for charger_id in group:
                    ratio = self.chargers[charger_id].requested_power / total_demand
                    allocated = available_w * ratio
                    allocation[charger_id] = allocated
                available_w = 0

        return allocation

    def update_charger_state(self, charger_id: str, is_charging: bool,
                              requested_power: float, soc: float = None):
        """
        更新充电桩状态
        """
        if charger_id in self.chargers:
            self.chargers[charger_id].is_charging = is_charging
            self.chargers[charger_id].requested_power = requested_power
            if soc is not None:
                self.chargers[charger_id].soc = soc

    def get_priority_allocation(self, charger_id: str) -> float:
        """
        获取优先级权重
        用于公平分配
        """
        state = self.chargers.get(charger_id)
        if not state:
            return 0

        # 优先级权重：VIP=3, 常规=2, 机会=1
        priority_weight = {1: 3, 2: 2, 3: 1}
        base_weight = priority_weight.get(state.priority, 1)

        # SOC权重：低SOC获得更高权重
        if state.soc is not None:
            soc_weight = (100 - state.soc) / 100  # SOC越低权重越高
        else:
            soc_weight = 0.5

        # 信用权重
        credit_weight = state.credit_weight

        return base_weight * 0.4 + soc_weight * 0.4 + credit_weight * 0.2


class ChargerState:
    """充电桩状态"""
    def __init__(self, charger_id: str, max_power: float, priority: int):
        self.charger_id = charger_id
        self.max_power = max_power
        self.priority = priority
        self.is_charging = False
        self.requested_power = 0
        self.soc = 50  # 默认SOC 50%
        self.credit_weight = 0.85  # 默认信用权重
        self.allocated_power = 0
        self.last_update = datetime.now()
```

```python
class V2GAggregator:
    """
    V2G聚合引擎
    将多个EV的V2G能力聚合为可调度的灵活性资源
    """
    def __init__(self):
        self.ev_fleet: Dict[str, EVState] = {}
        self.aggregated_capacity = {
            "charge_w": 0,
            "discharge_w": 0,
            "energy_kwh": 0
        }

    def register_ev(self, ev_id: str, battery_capacity_kwh: float,
                     max_charge_kw: float, max_discharge_kw: float):
        """
        注册EV到聚合池
        """
        self.ev_fleet[ev_id] = EVState(
            ev_id=ev_id,
            battery_capacity=battery_capacity_kwh,
            max_charge=max_charge_kw,
            max_discharge=max_discharge_kw
        )
        self._update_aggregated_capacity()

    def update_ev_state(self, ev_id: str, soc: float, is_connected: bool,
                         departure_time: Optional[datetime] = None):
        """
        更新EV状态
        """
        ev = self.ev_fleet.get(ev_id)
        if not ev:
            return

        ev.soc = soc
        ev.is_connected = is_connected
        if departure_time:
            ev.departure_time = departure_time

        self._update_aggregated_capacity()

    def calculate_available_flexibility(self) -> Dict:
        """
        计算可用灵活性容量
        """
        total_charge_w = 0
        total_discharge_w = 0
        total_energy_kwh = 0

        for ev_id, ev in self.ev_fleet.items():
            if not ev.is_connected:
                continue

            # 可充电容量（当前SOC < 目标SOC）
            if ev.soc < ev.target_soc:
                charge_power = ev.max_charge * 1000
                charge_energy = (ev.target_soc - ev.soc) / 100 * ev.battery_capacity
                total_charge_w += charge_power
                total_energy_kwh += charge_energy

            # 可放电容量（当前SOC > 最低SOC）
            if ev.soc > ev.min_soc:
                discharge_power = ev.max_discharge * 1000
                discharge_energy = (ev.soc - ev.min_soc) / 100 * ev.battery_capacity
                total_discharge_w += discharge_power

        return {
            "total_charge_capacity_w": total_charge_w,
            "total_discharge_capacity_w": total_discharge_w,
            "total_available_energy_kwh": total_energy_kwh,
            "ev_count": sum(1 for ev in self.ev_fleet.values() if ev.is_connected),
            "timestamp": int(datetime.now().timestamp() * 1000)
        }

    def dispatch_v2g(self, target_power_w: float, duration_s: int,
                      reason: str = "VPP_DISPATCH") -> Dict[str, float]:
        """
        调度V2G放电
        将目标功率分配到各EV
        """
        available = self.calculate_available_flexibility()
        if target_power_w > available["total_discharge_capacity_w"]:
            raise ValueError(f"Target power {target_power_w}W exceeds available capacity")

        # 按SOC比例分配放电功率
        total_available_soc = sum(
            ev.soc - ev.min_soc
            for ev in self.ev_fleet.values()
            if ev.is_connected and ev.soc > ev.min_soc
        )

        dispatch_plan = {}
        for ev_id, ev in self.ev_fleet.items():
            if not ev.is_connected or ev.soc <= ev.min_soc:
                dispatch_plan[ev_id] = 0
                continue

            # SOC越高，分配越多放电功率
            soc_ratio = (ev.soc - ev.min_soc) / total_available_soc if total_available_soc > 0 else 0
            allocated_power = target_power_w * soc_ratio

            # 不超过最大放电功率
            allocated_power = min(allocated_power, ev.max_discharge * 1000)
            dispatch_plan[ev_id] = allocated_power

        return dispatch_plan

    def _update_aggregated_capacity(self):
        """更新聚合容量"""
        total_charge = 0
        total_discharge = 0
        total_energy = 0

        for ev in self.ev_fleet.values():
            if ev.is_connected:
                total_charge += ev.max_charge * 1000
                total_discharge += ev.max_discharge * 1000
                total_energy += ev.battery_capacity * ev.soc / 100

        self.aggregated_capacity = {
            "charge_w": total_charge,
            "discharge_w": total_discharge,
            "energy_kwh": total_energy
        }
```

```python
class V2GScheduler:
    """
    V2G调度策略引擎
    根据电价、电网需求和用户偏好，优化V2G调度
    """
    def __init__(self, aggregator: V2GAggregator, price_client):
        self.aggregator = aggregator
        self.price_client = price_client
        self.schedule_mode = "AUTO"  # AUTO, MANUAL, VPP_FOLLOW

    def optimize_daily_schedule(self, price_forecast: List[Dict],
                                  pv_forecast: List[float]) -> Dict:
        """
        优化日间V2G调度计划
        """
        schedule = {
            "charge_periods": [],
            "discharge_periods": [],
            "idle_periods": []
        }

        for i, price_point in enumerate(price_forecast):
            price = price_point["price"]
            pv = pv_forecast[i] if i < len(pv_forecast) else 0

            # 低电价 + 光伏高峰 → 充电
            if price < 0.3 or pv > 5:
                schedule["charge_periods"].append(i)
            # 高电价 → 放电
            elif price > 0.8:
                schedule["discharge_periods"].append(i)
            else:
                schedule["idle_periods"].append(i)

        return schedule

    def respond_to_vpp_request(self, request_power_w: float,
                                duration_s: int) -> Dict:
        """
        响应VPP调度请求
        """
        # 检查可用容量
        available = self.aggregator.calculate_available_flexibility()

        if request_power_w > 0:  # 需要放电
            if request_power_w > available["total_discharge_capacity_w"]:
                return {"status": "REJECTED", "reason": "INSUFFICIENT_CAPACITY"}

            dispatch_plan = self.aggregator.dispatch_v2g(
                request_power_w, duration_s, reason="VPP_REQUEST"
            )
            return {
                "status": "ACCEPTED",
                "dispatch_plan": dispatch_plan,
                "total_power_w": request_power_w,
                "duration_s": duration_s
            }

        elif request_power_w < 0:  # 需要充电
            charge_power = abs(request_power_w)
            if charge_power > available["total_charge_capacity_w"]:
                return {"status": "REJECTED", "reason": "INSUFFICIENT_CAPACITY"}

            return {
                "status": "ACCEPTED",
                "charge_power_w": charge_power,
                "duration_s": duration_s
            }

        return {"status": "IGNORED"}

    def calculate_v2g_revenue(self, discharge_events: List[Dict]) -> Dict:
        """
        计算V2G放电收益
        """
        total_revenue = 0
        total_discharge_kwh = 0

        for event in discharge_events:
            power_kw = event["power_w"] / 1000
            duration_h = event["duration_s"] / 3600
            energy_kwh = power_kw * duration_h
            price = event.get("price_per_kwh", 0.8)

            revenue = energy_kwh * price
            total_revenue += revenue
            total_discharge_kwh += energy_kwh

        return {
            "total_discharge_kwh": total_discharge_kwh,
            "total_revenue": total_revenue,
            "avg_price_per_kwh": total_revenue / total_discharge_kwh if total_discharge_kwh > 0 else 0
        }
```

### 5.2 电池健康评估

```python
class BatteryHealthManager:
    """
    电池健康管理器
    监测EV电池SOH，优化充放电策略延长寿命
    """
    def __init__(self):
        self.battery_profiles: Dict[str, BatteryProfile] = {}

    def register_battery(self, ev_id: str, initial_capacity_kwh: float,
                          chemistry: str = "NMC"):
        """
        注册电池
        chemistry: NMC(三元锂), LFP(磷酸铁锂), LCO(钴酸锂)
        """
        self.battery_profiles[ev_id] = BatteryProfile(
            ev_id=ev_id,
            initial_capacity=initial_capacity_kwh,
            chemistry=chemistry
        )

    def record_cycle(self, ev_id: str, start_soc: float, end_soc: float,
                      avg_temp_c: float, avg_c_rate: float):
        """
        记录充放电循环
        """
        profile = self.battery_profiles.get(ev_id)
        if not profile:
            return

        # 计算循环深度
        dod = abs(end_soc - start_soc) / 100

        # 记录循环
        profile.cycles.append({
            "timestamp": datetime.now(),
            "start_soc": start_soc,
            "end_soc": end_soc,
            "dod": dod,
            "avg_temp": avg_temp_c,
            "avg_c_rate": avg_c_rate
        })

        # 更新循环次数（等效满充循环）
        profile.cycle_count += dod

        # 更新SOH
        profile.soh = self._estimate_soh(profile)

    def _estimate_soh(self, profile: BatteryProfile) -> float:
        """
        估算电池健康度（SOH）
        基于循环次数、温度、充放电倍率
        """
        # 基础衰减（每100次循环衰减约2%）
        base_degradation = profile.cycle_count / 100 * 0.02

        # 温度衰减（高温加速）
        temp_degradation = 0
        for cycle in profile.cycles[-100:]:  # 最近100次循环
            if cycle["avg_temp"] > 45:
                temp_degradation += 0.001  # 每次高温循环额外衰减0.1%
            elif cycle["avg_temp"] > 35:
                temp_degradation += 0.0005

        # 快充衰减（高C-rate加速）
        fast_charge_degradation = 0
        for cycle in profile.cycles[-100:]:
            if cycle["avg_c_rate"] > 1.5:
                fast_charge_degradation += 0.0008
            elif cycle["avg_c_rate"] > 1.0:
                fast_charge_degradation += 0.0003

        # 总衰减
        total_degradation = base_degradation + temp_degradation + fast_charge_degradation
        soh = max(0.7, 1.0 - total_degradation)  # SOH不低于70%

        return soh

    def get_optimal_charging_strategy(self, ev_id: str) -> Dict:
        """
        获取最优充电策略
        基于电池健康状态优化充放电参数
        """
        profile = self.battery_profiles.get(ev_id)
        if not profile:
            return {}

        strategy = {
            "optimal_charge_range": [20, 80],  # 最优SOC区间
            "max_charge_c_rate": 1.0,  # 最大充电倍率
            "max_discharge_c_rate": 1.0,  # 最大放电倍率
            "optimal_temp_range": [15, 35],  # 最优温度区间
            "avoid_deep_discharge": True,  # 避免深度放电
            "prefer_slow_charge": True  # 优先慢充
        }

        # 根据SOH调整策略
        if profile.soh < 0.85:
            strategy["optimal_charge_range"] = [30, 70]  # 缩小SOC区间
            strategy["max_charge_c_rate"] = 0.8  # 降低充电倍率
            strategy["max_discharge_c_rate"] = 0.8

        if profile.soh < 0.75:
            strategy["optimal_charge_range"] = [40, 60]
            strategy["max_charge_c_rate"] = 0.5
            strategy["max_discharge_c_rate"] = 0.5
            strategy["avoid_v2g"] = True  # 避免V2G放电

        return strategy

    def calculate_shadow_depreciation(self, ev_id: str) -> Dict:
        """
        计算影子折旧
        将充放电操作对电池寿命的影响量化为成本
        """
        profile = self.battery_profiles.get(ev_id)
        if not profile:
            return {}

        # 电池更换成本（假设）
        replacement_cost = 80000  # 80kWh电池更换成本

        # 当前SOH对应的折旧
        soh_loss = 1.0 - profile.soh
        depreciation = soh_loss * replacement_cost

        # 最近100次循环的折旧明细
        recent_depreciation = []
        for cycle in profile.cycles[-100:]:
            cycle_cost = self._calculate_cycle_cost(cycle, replacement_cost)
            recent_depreciation.append({
                "timestamp": cycle["timestamp"],
                "dod": cycle["dod"],
                "cost": cycle_cost
            })

        return {
            "ev_id": ev_id,
            "current_soh": profile.soh,
            "total_cycles": profile.cycle_count,
            "total_depreciation": depreciation,
            "remaining_value": replacement_cost - depreciation,
            "recent_depreciation": recent_depreciation
        }

    def _calculate_cycle_cost(self, cycle: Dict, replacement_cost: float) -> float:
        """计算单次循环的折旧成本"""
        # 假设电池寿命为1000次等效满充循环
        total_cycles_lifetime = 1000
        cost_per_full_cycle = replacement_cost / total_cycles_lifetime
        return cost_per_full_cycle * cycle["dod"]


class BatteryProfile:
    """电池档案"""
    def __init__(self, ev_id: str, initial_capacity: float, chemistry: str):
        self.ev_id = ev_id
        self.initial_capacity = initial_capacity  # kWh
        self.current_capacity = initial_capacity
        self.chemistry = chemistry
        self.soh = 1.0  # 健康度
        self.cycle_count = 0  # 等效满充循环次数
        self.cycles: List[Dict] = []  # 循环记录
        self.manufacture_date = datetime.now()
        self.warranty_cycles = 1000  # 保修循环次数
```

```python
class BatteryHealthOptimizer:
    """
    电池健康调度优化器
    在VPP收益和电池寿命之间寻找最优平衡
    """
    def __init__(self, health_manager: BatteryHealthManager):
        self.health_manager = health_manager

    def optimize_v2g_dispatch(self, ev_id: str, requested_power_w: float,
                               price_per_kwh: float) -> Dict:
        """
        优化V2G调度决策
        考虑电池健康成本
        """
        profile = self.health_manager.battery_profiles.get(ev_id)
        if not profile:
            return {"dispatch": False, "reason": "UNKNOWN_EV"}

        strategy = self.health_manager.get_optimal_charging_strategy(ev_id)

        # 检查是否允许V2G
        if strategy.get("avoid_v2g", False):
            return {"dispatch": False, "reason": "BATTERY_HEALTH_TOO_LOW"}

        # 计算V2G收益
        power_kw = requested_power_w / 1000
        duration_h = 1  # 假设1小时
        energy_kwh = power_kw * duration_h
        revenue = energy_kwh * price_per_kwh

        # 计算电池折旧成本
        depreciation = self.health_manager.calculate_shadow_depreciation(ev_id)
        cycle_cost = depreciation["total_depreciation"] / max(1, profile.cycle_count)
        estimated_cycle_cost = cycle_cost * (energy_kwh / profile.initial_capacity)

        # 净收益
        net_profit = revenue - estimated_cycle_cost

        return {
            "dispatch": net_profit > 0,
            "revenue": revenue,
            "battery_cost": estimated_cycle_cost,
            "net_profit": net_profit,
            "soh_after": profile.soh - 0.001,  # 估算SOH下降
            "recommendation": "DISPATCH" if net_profit > 0 else "HOLD"
        }
```

```python
class ChargingBillingEngine:
    """
    充电计费引擎
    支持分时计费、会员计费、V2G收益结算
    """
    def __init__(self):
        self.price_tables: Dict[str, PriceTable] = {}
        self.member_tiers: Dict[str, float] = {
            "NORMAL": 1.0,    # 普通会员：原价
            "SILVER": 0.9,    # 银卡会员：9折
            "GOLD": 0.8,      # 金卡会员：8折
            "PLATINUM": 0.7   # 铂金会员：7折
        }

    def register_price_table(self, station_id: str, price_table: 'PriceTable'):
        """注册充电站电价表"""
        self.price_tables[station_id] = price_table

    def calculate_charging_cost(self, station_id: str, energy_kwh: float,
                                 duration_h: float, member_tier: str = "NORMAL",
                                 start_time: Optional[datetime] = None) -> Dict:
        """
        计算充电费用
        """
        price_table = self.price_tables.get(station_id)
        if not price_table:
            return {"error": "PRICE_TABLE_NOT_FOUND"}

        # 获取电价
        if start_time is None:
            start_time = datetime.now()

        price_per_kwh = price_table.get_price(start_time)
        service_fee = price_table.service_fee_per_kwh

        # 电费
        energy_cost = energy_kwh * price_per_kwh

        # 服务费
        service_cost = energy_kwh * service_fee

        # 会员折扣
        discount = self.member_tiers.get(member_tier, 1.0)
        total_cost = (energy_cost + service_cost) * discount

        return {
            "station_id": station_id,
            "energy_kwh": energy_kwh,
            "duration_h": duration_h,
            "price_per_kwh": price_per_kwh,
            "service_fee_per_kwh": service_fee,
            "energy_cost": energy_cost,
            "service_cost": service_cost,
            "member_discount": (1 - discount) * 100,
            "total_cost": total_cost,
            "member_tier": member_tier
        }

    def calculate_v2g_payment(self, ev_id: str, discharge_kwh: float,
                               price_per_kwh: float) -> Dict:
        """
        计算V2G放电收益
        """
        payment = discharge_kwh * price_per_kwh

        return {
            "ev_id": ev_id,
            "discharge_kwh": discharge_kwh,
            "price_per_kwh": price_per_kwh,
            "payment": payment,
            "payment_time": datetime.now()
        }


class PriceTable:
    """电价表"""
    def __init__(self, station_id: str, service_fee_per_kwh: float = 0.1):
        self.station_id = station_id
        self.service_fee_per_kwh = service_fee_per_kwh
        self.time_periods: List[Dict] = []

    def add_time_period(self, start_hour: int, end_hour: int,
                         price_per_kwh: float, days: List[str] = None):
        """
        添加时段电价
        days: ["weekday", "weekend"] 或 None（全部）
        """
        self.time_periods.append({
            "start_hour": start_hour,
            "end_hour": end_hour,
            "price": price_per_kwh,
            "days": days or ["all"]
        })

    def get_price(self, time_point: datetime) -> float:
        """
        获取指定时间的电价
        """
        hour = time_point.hour
        weekday = time_point.weekday()

        for period in self.time_periods:
            days = period["days"]
            # 检查日期匹配
            if "weekday" in days and weekday >= 5:
                continue
            if "weekend" in days and weekday < 5:
                continue

            # 检查小时匹配
            if period["start_hour"] <= hour < period["end_hour"]:
                return period["price"]

        # 默认电价
        return 0.5
```

```python
class ChargingStationManager:
    """
    充电站容量管理器
    管理充电站的总容量和负载分配
    """

        self.transformer_capacity = transformer_capacity_kva
        self.chargers: Dict[str, ChargerInfo] = {}
        self.dlb = DynamicLoadBalancer(transformer_capacity_kva * 1000)

    def add_charger(self, charger_id: str, max_power_kw: float,
                     charger_type: str = "AC"):
        """
        添加充电桩
        """
        self.chargers[charger_id] = ChargerInfo(
            charger_id=charger_id,
            max_power=max_power_kw,
            charger_type=charger_type
        )
        self.dlb.register_charger(charger_id, max_power_kw * 1000)

    def get_station_status(self) -> Dict:
        """
        获取充电站状态
        """
        total_capacity = sum(c.max_power for c in self.chargers.values())
        current_load = sum(c.current_power for c in self.chargers.values() if c.is_charging)
        available = total_capacity - current_load

        return {
            "station_id": self.station_id,
            "transformer_capacity_kva": self.transformer_capacity,
            "total_charger_capacity_kw": total_capacity,
            "current_load_kw": current_load,
            "available_capacity_kw": available,
            "utilization_pct": (current_load / total_capacity * 100) if total_capacity > 0 else 0,
            "online_chargers": sum(1 for c in self.chargers.values() if c.is_online),
            "charging_evs": sum(1 for c in self.chargers.values() if c.is_charging),
            "fault_chargers": sum(1 for c in self.chargers.values() if c.has_fault)
        }
```
