### 1.2 四川电力市场特征

四川电力市场具有显著的**丰枯水期差异**和**跨省互联**特征，是本方案的核心建模对象：

| 特征               |   丰水期（6-10月）    |   枯水期（11-5月）   | 对交易策略的影响           |
| ------------------ | :-------------------: | :------------------: | -------------------------- |
| **水电出力**       |  充沛（占全省80%+）   | 不足（占全省40-50%） | 丰水期电价低，枯水期电价高 |
| **平均电价**       |    0.15-0.30元/kWh    |   0.35-0.60元/kWh    | 价差可达2-3倍              |
| **尖峰电价**       |    0.50-0.80元/kWh    |   0.80-1.50元/kWh    | 枯水期尖峰时段需重点管理   |
| **跨省外送**       | 大量外送（华东/西北） |    少量外送或受入    | 丰水期外送套利，枯水期购入 |
| **联络线可用容量** |    高（水电满发）     |    低（水电受限）    | 影响省间交易可行性         |
| **绿电供应**       |   充足（水电+光伏）   |  不足（需外购绿电）  | 影响碳资产管理策略         |

### 1.3 四川丰枯水期电价建模

```python
class SichuanSeasonalPricing:
    """
    四川丰枯水期差异电价模型
    将季节性电价特征转化为数学模型，用于交易决策
    """
    def __init__(self):
        # 丰水期定义（6月1日 - 10月31日）
        self.wet_season_start = (6, 1)
        self.wet_season_end = (10, 31)

        # 枯水期定义（11月1日 - 5月31日）
        self.dry_season_start = (11, 1)
        self.dry_season_end = (5, 31)

        # 丰枯水期基准电价（元/kWh）
        self.base_prices = {
            "wet": {
                "valley": 0.12,    # 低谷（23:00-7:00）
                "flat": 0.20,      # 平段（7:00-10:00, 15:00-18:00, 21:00-23:00）
                "peak": 0.35,      # 高峰（10:00-12:00, 14:00-15:00, 18:00-21:00）
                "sharp": 0.55      # 尖峰（12:00-14:00, 夏季特定时段）
            },
            "dry": {
                "valley": 0.25,
                "flat": 0.40,
                "peak": 0.65,
                "sharp": 1.20
            }
        }

        # 尖峰时段动态调整（基于天气预报和电网负荷预测）
        self.sharp_peak_adjustment = {
            "summer_extreme": 1.3,   # 夏季极端高温日尖峰电价上浮30%
            "winter_extreme": 1.2,   # 冬季极端低温日尖峰电价上浮20%
            "normal": 1.0            # 正常日
        }

    def get_season(self, date: datetime) -> str:
        """
        判断日期所属季节
        返回: "wet"（丰水期）或 "dry"（枯水期）
        """
        month = date.month
        day = date.day

        if (month == self.wet_season_start[0] and day >= self.wet_season_start[1]) or \
           (self.wet_season_start[0] < month < self.wet_season_end[0]) or \
           (month == self.wet_season_end[0] and day <= self.wet_season_end[1]):
            return "wet"
        return "dry"

    def get_period(self, hour: int) -> str:
        """
        获取时段类型
        返回: "valley", "flat", "peak", "sharp"
        """
        if 23 <= hour or hour < 7:
            return "valley"
        elif 7 <= hour < 10 or 15 <= hour < 18 or 21 <= hour < 23:
            return "flat"
        elif 10 <= hour < 12 or 14 <= hour < 15 or 18 <= hour < 21:
            return "peak"
        elif 12 <= hour < 14:
            return "sharp"
        return "flat"

    def get_base_price(self, date: datetime, hour: int) -> float:
        """
        获取基准电价
        """
        season = self.get_season(date)
        period = self.get_period(hour)
        return self.base_prices[season][period]

    def get_adjusted_price(self, date: datetime, hour: int,
                           weather_factor: str = "normal") -> float:
        """
        获取调整后的电价（考虑尖峰时段动态调整）
        """
        base_price = self.get_base_price(date, hour)
        period = self.get_period(hour)

        if period == "sharp":
            adjustment = self.sharp_peak_adjustment.get(weather_factor, 1.0)
            return base_price * adjustment

        return base_price

    def get_daily_price_curve(self, date: datetime,
                               weather_factor: str = "normal") -> List[float]:
        """
        获取日电价曲线（96个15分钟时段）
        """
        curve = []
        for period in range(96):
            hour = period // 4
            price = self.get_adjusted_price(date, hour, weather_factor)
            curve.append(price)
        return curve

    def calculate_arbitrage_spread(self, date: datetime) -> Dict:
        """
        计算日内最大套利价差
        用于储能充放电策略
        """
        prices = self.get_daily_price_curve(date)
        max_price = max(prices)
        min_price = min(prices)
        max_spread = max_price - min_price

        # 找到最优充放电时段
        charge_hour = prices.index(min_price) // 4
        discharge_hour = prices.index(max_price) // 4

        return {
            "date": date,
            "season": self.get_season(date),
            "max_price": max_price,
            "min_price": min_price,
            "max_spread": max_spread,
            "optimal_charge_hour": charge_hour,
            "optimal_discharge_hour": discharge_hour,
            "arbitrage_revenue_per_kwh": max_spread * 0.9  # 考虑效率损失
        }
```

### 1.4 跨省联络线建模

四川电网通过多条特高压线路与西北、华东电网互联，跨省交易是本方案的核心能力：

| 联络线         | 起点 | 终点 |  容量  | 方向      | 交易机会                    |
| -------------- | ---- | ---- | :----: | --------- | --------------------------- |
| **陇电入川**   | 甘肃 | 四川 | 8000MW | 甘肃→四川 | 枯水期购入甘肃低价火电/风电 |
| **川苏特高压** | 四川 | 江苏 | 7200MW | 四川→华东 | 丰水期外送四川水电          |
| **川渝联络线** | 四川 | 重庆 | 3000MW | 双向      | 川渝互济                    |
| **德宝直流**   | 四川 | 陕西 | 3000MW | 双向      | 西北-西南互济               |

```python
class InterprovincialTieLine:
    """
    跨省联络线模型
    用于省间现货交易优化
    """
    def __init__(self, line_id: str, from_province: str, to_province: str,
                 capacity_mw: float, wheeling_charge_per_kwh: float):
        self.line_id = line_id
        self.from_province = from_province
        self.to_province = to_province
        self.capacity = capacity_mw  # MW
        self.wheeling_charge = wheeling_charge_per_kwh  # 输电网费 元/kWh
        self.available_atc = capacity_mw  # 可用传输容量
        self.current_flow = 0  # 当前潮流

    def update_atc(self, scheduled_flow_mw: float,
                   maintenance_factor: float = 1.0):
        """
        更新可用传输容量（ATC）
        考虑计划潮流和维护因素
        """
        self.available_atc = (self.capacity - scheduled_flow_mw) * maintenance_factor
        return self.available_atc

    def check_feasibility(self, import_power_mw: float) -> bool:
        """
        检查跨省购电是否可行
        """
        return import_power_mw <= self.available_atc

    def calculate_total_cost(self, energy_kwh: float,
                              source_price_per_kwh: float) -> float:
        """
        计算跨省购电总成本
        总成本 = 电量×源端电价 + 电量×输电网费
        """
        energy_cost = energy_kwh * source_price_per_kwh
        transmission_cost = energy_kwh * self.wheeling_charge
        return energy_cost + transmission_cost


class SichuanInterprovincialNetwork:
    """
    四川跨省联络线网络
    管理所有联络线的状态和交易优化
    """
    def __init__(self):
        self.tie_lines = {
            "longdian_into_chuan": InterprovincialTieLine(
                "LD-IC", "甘肃", "四川", 8000, 0.05
            ),
            "chuan_su_uHVDC": InterprovincialTieLine(
                "CS-UHV", "四川", "江苏", 7200, 0.08
            ),
            "chuan_yu": InterprovincialTieLine(
                "CY", "四川", "重庆", 3000, 0.02
            ),
            "de_bao_DC": InterprovincialTieLine(
                "DB-DC", "四川", "陕西", 3000, 0.03
            )
        }

    def evaluate_arbitrage_opportunity(self, date: datetime,
                                        sichuan_price: float,
                                        gansu_price: float) -> Dict:
        """
        评估川-甘跨省套利机会
        枯水期四川电价高时，从甘肃购入低价电力
        """
        tie_line = self.tie_lines["longdian_into_chuan"]
        season = SichuanSeasonalPricing().get_season(date)

        # 价差 = 四川电价 - (甘肃电价 + 输电网费)
        total_gansu_cost = gansu_price + tie_line.wheeling_charge
        spread = sichuan_price - total_gansu_cost

        opportunity = {
            "date": date,
            "season": season,
            "sichuan_price": sichuan_price,
            "gansu_price": gansu_price,
            "wheeling_charge": tie_line.wheeling_charge,
            "total_gansu_cost": total_gansu_cost,
            "spread": spread,
            "is_profitable": spread > 0.02,  # 至少2分/kWh利润
            "available_capacity_mw": tie_line.available_atc,
            "recommendation": "IMPORT" if spread > 0.02 else "HOLD"
        }

        return opportunity

    def optimize_interprovincial_dispatch(self, date: datetime,
                                           local_load_mw: float,
                                           local_pv_mw: float,
                                           local_hydro_mw: float,
                                           price_forecast: Dict) -> Dict:
        """
        优化跨省调度决策
        综合考虑本地供需、联络线容量和电价
        """
        # 计算本地净负荷
        net_load = local_load_mw - local_pv_mw - local_hydro_mw

        dispatch_plan = {
            "local_generation": 0,
            "import_from_gansu": 0,
            "export_to_jiangsu": 0,
            "total_cost": 0
        }

        # 如果本地净负荷为正（需要购电）
        if net_load > 0:
            # 评估从甘肃购电
            gansu_opportunity = self.evaluate_arbitrage_opportunity(
                date, price_forecast["sichuan"], price_forecast["gansu"]
            )

            if gansu_opportunity["is_profitable"]:
                import_power = min(net_load,
                                   self.tie_lines["longdian_into_chuan"].available_atc)
                dispatch_plan["import_from_gansu"] = import_power
                dispatch_plan["local_generation"] = net_load - import_power
            else:
                dispatch_plan["local_generation"] = net_load

        # 如果本地净负荷为负（电力过剩）
        elif net_load < 0:
            surplus = abs(net_load)
            # 评估外送江苏
            export_power = min(surplus,
                               self.tie_lines["chuan_su_uHVDC"].available_atc)
            dispatch_plan["export_to_jiangsu"] = export_power

        return dispatch_plan
```

### 2.2 基于epftoolbox的电价预测

```python
import numpy as np
import pandas as pd
from datetime import datetime, timedelta
from typing import List, Dict, Optional, Tuple
import warnings
warnings.filterwarnings('ignore')


class EPFToolboxPredictor:
    """
    基于epftoolbox的电价预测引擎
    支持四川电力市场的特殊特征建模
    """
    def __init__(self):
        self.models = {}
        self.scaler = None
        self.feature_columns = []
        self.is_trained = False

        # 四川特殊特征
        self.seasonal_pricing = SichuanSeasonalPricing()
        self.sichuan_features = [
            "hour", "day_of_week", "month", "is_weekend", "is_holiday",
            "season_wet", "season_dry",
            "lag_1_price", "lag_2_price", "lag_24_price", "lag_168_price",
            "rolling_mean_24h", "rolling_std_24h",
            "temperature", "precipitation", "humidity",
            "total_load", "hydro_output", "pv_output",
            "tie_line_flow_gansu", "tie_line_flow_jiangsu",
            "reservoir_level", "inflow_forecast"
        ]

    def prepare_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        特征工程
        构建电价预测所需的全部特征
        """
        features = df.copy()

        # 时间特征
        features['hour'] = features.index.hour
        features['day_of_week'] = features.index.dayofweek
        features['month'] = features.index.month
        features['is_weekend'] = features['day_of_week'].isin([5, 6]).astype(int)
        features['is_holiday'] = self._is_holiday(features.index)

        # 四川丰枯水期特征
        features['season_wet'] = features.index.map(
            lambda x: 1 if self.seasonal_pricing.get_season(x) == "wet" else 0
        )
        features['season_dry'] = features.index.map(
            lambda x: 1 if self.seasonal_pricing.get_season(x) == "dry" else 0
        )

        # 滞后特征
        features['lag_1_price'] = features['price'].shift(1)
        features['lag_2_price'] = features['price'].shift(2)
        features['lag_24_price'] = features['price'].shift(24 * 12)  # 24小时前（5min粒度）
        features['lag_168_price'] = features['price'].shift(168 * 12)  # 一周前

        # 滚动统计
        features['rolling_mean_24h'] = features['price'].rolling(
            window=24 * 12, min_periods=1
        ).mean()
        features['rolling_std_24h'] = features['price'].rolling(
            window=24 * 12, min_periods=1
        ).std()

        # 填充缺失值
        features = features.fillna(method='bfill').fillna(method='ffill')

        return features

    def build_lstm_model(self, seq_length: int = 24 * 12) -> 'tf.keras.Model':
        """
        构建LSTM电价预测模型
        使用TensorFlow/Keras实现
        """
        from tensorflow.keras.models import Sequential
        from tensorflow.keras.layers import LSTM, Dense, Dropout
        from tensorflow.keras.optimizers import Adam

        model = Sequential([
            LSTM(128, return_sequences=True, input_shape=(seq_length, len(self.feature_columns))),
            Dropout(0.2),
            LSTM(64, return_sequences=False),
            Dropout(0.2),
            Dense(32, activation='relu'),
            Dense(1)  # 输出：下一个时段的电价
        ])

        model.compile(
            optimizer=Adam(learning_rate=0.001),
            loss='mae',
            metrics=['mae', 'mse']
        )

        return model

    def build_transformer_model(self) -> 'tf.keras.Model':
        """
        构建Transformer电价预测模型
        使用注意力机制捕捉长程依赖
        """
        from tensorflow.keras.models import Model
        from tensorflow.keras.layers import Input, MultiHeadAttention, LayerNormalization, GlobalAveragePooling1D, Dense

        inputs = Input(shape=(None, len(self.feature_columns)))

        # Transformer编码器层
        attention = MultiHeadAttention(num_heads=8, key_dim=64)(inputs, inputs)
        attention = LayerNormalization()(attention + inputs)

        # 前馈网络
        ff = Dense(256, activation='relu')(attention)
        ff = Dense(len(self.feature_columns))(ff)
        ff = LayerNormalization()(ff + attention)

        # 全局池化 + 输出
        pooled = GlobalAveragePooling1D()(ff)
        outputs = Dense(1)(pooled)

        model = Model(inputs=inputs, outputs=outputs)
        model.compile(optimizer='adam', loss='mae', metrics=['mae'])

        return model

    def train(self, historical_data: pd.DataFrame,
              target_column: str = 'price',
              validation_split: float = 0.2):
        """
        训练电价预测模型
        """
        # 特征工程
        features = self.prepare_features(historical_data)
        self.feature_columns = [col for col in features.columns
                                if col != target_column]

        # 准备训练数据
        X = features[self.feature_columns].values
        y = features[target_column].values

        # 归一化
        from sklearn.preprocessing import StandardScaler
        self.scaler = StandardScaler()
        X_scaled = self.scaler.fit_transform(X)

        # 构建LSTM模型
        self.models['lstm'] = self.build_lstm_model()
        self.models['transformer'] = self.build_transformer_model()

        # 训练LSTM
        print("Training LSTM model...")
        X_lstm = X_scaled.reshape((X_scaled.shape[0], 1, X_scaled.shape[1]))
        self.models['lstm'].fit(
            X_lstm, y,
            validation_split=validation_split,
            epochs=50,
            batch_size=32,
            verbose=1
        )

        # 训练Transformer
        print("Training Transformer model...")
        self.models['transformer'].fit(
            X_scaled.reshape((X_scaled.shape[0], 1, X_scaled.shape[1])), y,
            validation_split=validation_split,
            epochs=30,
            batch_size=32,
            verbose=1
        )

        self.is_trained = True
        print("Model training completed")

    def predict(self, input_features: pd.DataFrame,
                use_ensemble: bool = True) -> np.ndarray:
        """
        电价预测
        支持单模型预测和集成预测
        """
        if not self.is_trained:
            raise ValueError("Model not trained yet")

        # 特征准备
        features = self.prepare_features(input_features)
        X = features[self.feature_columns].values

        # 归一化
        X_scaled = self.scaler.transform(X)

        if use_ensemble:
            # 集成预测（加权平均）
            predictions = []
            weights = {'lstm': 0.5, 'transformer': 0.5}

            for name, model in self.models.items():
                X_reshaped = X_scaled.reshape((X_scaled.shape[0], 1, X_scaled.shape[1]))
                pred = model.predict(X_reshaped, verbose=0).flatten()
                predictions.append(pred * weights[name])

            ensemble_pred = np.sum(predictions, axis=0)
            return ensemble_pred
        else:
            # 使用LSTM模型
            X_reshaped = X_scaled.reshape((X_scaled.shape[0], 1, X_scaled.shape[1]))
            return self.models['lstm'].predict(X_reshaped, verbose=0).flatten()

    def predict_with_uncertainty(self, input_features: pd.DataFrame,
                                  n_iterations: int = 100) -> Dict:
        """
        带不确定性的电价预测
        通过MC Dropout生成预测分布
        """
        predictions = []

        for _ in range(n_iterations):
            pred = self.predict(input_features, use_ensemble=True)
            predictions.append(pred)

        predictions = np.array(predictions)

        return {
            "mean": np.mean(predictions, axis=0),
            "std": np.std(predictions, axis=0),
            "p10": np.percentile(predictions, 10, axis=0),
            "p50": np.percentile(predictions, 50, axis=0),
            "p90": np.percentile(predictions, 90, axis=0),
            "prediction_interval": {
                "lower": np.percentile(predictions, 10, axis=0),
                "upper": np.percentile(predictions, 90, axis=0)
            }
        }

    def predict_sichuan_da_price(self, date: datetime,
                                   weather_forecast: Dict,
                                   load_forecast: Dict,
                                   hydro_forecast: Dict) -> Dict:
        """
        预测四川日前市场电价
        融合丰枯水期特征和来水预测
        """
        season = self.seasonal_pricing.get_season(date)

        # 构建输入特征
        hours = list(range(24))
        prices = []

        for hour in hours:
            # 基础电价（丰枯水期基准）
            base_price = self.seasonal_pricing.get_base_price(date, hour)

            # 天气预报修正
            temp = weather_forecast.get('temperature', [25])[hour]
            if temp > 35:  # 高温导致空调负荷增加
                base_price *= 1.15
            elif temp < 5:  # 低温导致取暖负荷增加
                base_price *= 1.10

            # 来水预测修正（丰水期来水多→电价低）
            if season == "wet":
                inflow = hydro_forecast.get('inflow', 1.0)
                if inflow > 1.2:  # 来水偏多
                    base_price *= 0.9
                elif inflow < 0.8:  # 来水偏少
                    base_price *= 1.1

            # 负荷预测修正
            peak_load = load_forecast.get('peak_load_mw', 0)
            if peak_load > 50000:  # 全省负荷高峰
                base_price *= 1.2

            prices.append(base_price)

        return {
            "date": date,
            "season": season,
            "forecast_prices": prices,
            "min_price": min(prices),
            "max_price": max(prices),
            "avg_price": np.mean(prices),
            "volatility": np.std(prices) / np.mean(prices)
        }

            if key in holidays:

        return is_holiday
```
