## 2. 现货市场结算

### 2.1 结算规则

四川电力现货市场采用"日前市场+实时市场"双结算模式：

| 结算项目         | 计算方式                   | 说明                 |
| ---------------- | -------------------------- | -------------------- |
| **日前市场结算** | 日前中标电量 × 日前出清价  | 按15分钟时段结算     |
| **实时市场结算** | 实时偏差电量 × 实时出清价  | 偏差 = 实际 - 日前   |
| **不平衡结算**   | 偏差电量 × 偏差价格        | 超出免考核范围的部分 |
| **辅助服务结算** | 调频/调峰中标容量 × 出清价 | 按小时结算           |

### 2.2 结算计算引擎

```python
from decimal import Decimal, ROUND_HALF_UP
from datetime import datetime, timedelta
from typing import List, Dict, Optional
import pandas as pd


class SpotSettlementEngine:
    """
    现货市场结算引擎
    使用Decimal确保财务精度
    """
    def __init__(self):
        # 精度配置
        self.precision = Decimal('0.000001')  # 小数点后6位
        self.rounding = ROUND_HALF_UP

    def calculate_period_settlement(self,
                                      scheduled_energy: Decimal,
                                      actual_energy: Decimal,
                                      da_price: Decimal,  # 日前价格
                                      rt_price: Decimal,  # 实时价格
                                      free_tolerance: Decimal = Decimal('0.03')) -> Dict:
        """
        计算单个时段的现货结算
        """
        # 日前结算
        da_settlement = scheduled_energy * da_price

        # 偏差电量
        deviation = actual_energy - scheduled_energy

        # 实时结算（偏差部分）
        rt_settlement = deviation * rt_price

        # 总结算金额
        total_settlement = da_settlement + rt_settlement

        # 四舍五入到6位小数
        return {
            "scheduled_energy": scheduled_energy.quantize(self.precision),
            "actual_energy": actual_energy.quantize(self.precision),
            "deviation": deviation.quantize(self.precision),
            "da_price": da_price.quantize(self.precision),
            "rt_price": rt_price.quantize(self.precision),
            "da_settlement": da_settlement.quantize(self.precision),
            "rt_settlement": rt_settlement.quantize(self.precision),
            "total_settlement": total_settlement.quantize(self.precision)
        }

    def calculate_daily_settlement(self, periods: List[Dict]) -> Dict:
        """
        计算日结算汇总
        """
        total_da = Decimal('0')
        total_rt = Decimal('0')
        total_deviation = Decimal('0')
        total_energy = Decimal('0')

        for period in periods:
            result = self.calculate_period_settlement(
                Decimal(str(period['scheduled_energy'])),
                Decimal(str(period['actual_energy'])),
                Decimal(str(period['da_price'])),
                Decimal(str(period['rt_price']))
            )
            total_da += result['da_settlement']
            total_rt += result['rt_settlement']
            total_deviation += result['deviation']
            total_energy += result['actual_energy']

        return {
            "total_da_settlement": total_da.quantize(self.precision),
            "total_rt_settlement": total_rt.quantize(self.precision),
            "total_settlement": (total_da + total_rt).quantize(self.precision),
            "total_energy_mwh": total_energy.quantize(self.precision),
            "total_deviation_mwh": total_deviation.quantize(self.precision),
            "average_da_price": (total_da / total_energy).quantize(self.precision) if total_energy > 0 else Decimal('0'),
            "period_count": len(periods)
        }

    def validate_settlement(self, settlement: Dict,
                             trading_center_settlement: Dict) -> List[str]:
        """
        校验结算结果与交易中心结算单的一致性
        """
        discrepancies = []

        # 比较总金额
        our_total = settlement['total_settlement']
        their_total = Decimal(str(trading_center_settlement['total_amount']))

        diff = abs(our_total - their_total)
        if diff > Decimal('0.01'):  # 允许1分钱误差
            discrepancies.append(
                f"Total settlement discrepancy: our={our_total}, theirs={their_total}, diff={diff}"
            )

        # 比较各时段
        for period in settlement.get('periods', []):
            their_period = trading_center_settlement.get('periods', {}).get(period['period_id'])
            if their_period:
                period_diff = abs(
                    Decimal(str(period['total_settlement'])) -
                    Decimal(str(their_period['amount']))
                )
                if period_diff > Decimal('0.001'):
                    discrepancies.append(
                        f"Period {period['period_id']} discrepancy: {period_diff}"
                    )

        return discrepancies
```

| ---------------- | -------------------- | ------------------ | ----------------- |

### 3.2 合同结算引擎

```python
class ContractSettlementEngine:
    """
    中长期合同结算引擎
    """
    def __init__(self):
        self.precision = Decimal('0.000001')

    def calculate_contract_settlement(self,
                                        contract: Dict,
                                        actual_execution: List[Dict]) -> Dict:
        """
        计算合同结算
        """
        contract_id = contract['contract_id']
        contract_type = contract['contract_type']
        contracted_energy = Decimal(str(contract['contracted_energy_mwh']))
        contract_price = Decimal(str(contract['price_per_mwh']))

        # 计算实际执行电量
        total_executed = sum(
            Decimal(str(p['executed_energy_mwh'])) for p in actual_execution
        )

        # 计算偏差
        deviation = total_executed - contracted_energy
        deviation_pct = (deviation / contracted_energy * 100) if contracted_energy > 0 else Decimal('0')

        # 获取免考核范围
        tolerance = self._get_tolerance(contract_type)

        # 计算考核费用
        penalty = Decimal('0')
        if abs(deviation_pct) > tolerance:
            excess_deviation = abs(deviation) - contracted_energy * tolerance / Decimal('100')
            penalty_rate = self._get_penalty_rate(contract_type, deviation > 0)
            penalty = excess_deviation * contract_price * penalty_rate

        # 结算金额
        base_settlement = total_executed * contract_price
        total_settlement = base_settlement - penalty

        return {
            "contract_id": contract_id,
            "contract_type": contract_type,
            "contracted_energy": contracted_energy.quantize(self.precision),
            "executed_energy": total_executed.quantize(self.precision),
            "deviation_mwh": deviation.quantize(self.precision),
            "deviation_pct": deviation_pct.quantize(Decimal('0.01')),
            "contract_price": contract_price.quantize(self.precision),
            "base_settlement": base_settlement.quantize(self.precision),
            "deviation_penalty": penalty.quantize(self.precision),
            "total_settlement": total_settlement.quantize(self.precision)
        }

    def _get_tolerance(self, contract_type: str) -> Decimal:
        """获取免考核偏差范围"""
        tolerances = {
            'ANNUAL': Decimal('5.0'),
            'MONTHLY': Decimal('3.0'),
            'WEEKLY': Decimal('2.0'),
            'GREEN': Decimal('10.0')
        }
        return tolerances.get(contract_type, Decimal('3.0'))

    def _get_penalty_rate(self, contract_type: str, is_over: bool) -> Decimal:
        """获取考核系数"""
        if is_over:
            return Decimal('0.8')  # 超发考核系数
        return Decimal('1.2')  # 欠发考核系数
```

## 4. 偏差考核管理

### 4.1 偏差考核规则

四川电力市场偏差考核规则：

| 偏差类型          | 免考核范围  |              考核价格               | 说明       |
| ----------------- | :---------: | :---------------------------------: | ---------- |
| **日前-实时偏差** |     ±3%     | 实时出清价×1.2（欠发）/×0.8（超发） | 15分钟结算 |
| **合同执行偏差**  | ±5%（月度） |   合同价×1.2（欠发）/×0.8（超发）   | 月度考核   |
| **辅助服务偏差**  |     ±2%     |             出清价×2.0              | 小时考核   |
| **联络线偏差**    |     ±1%     |           省间现货价×1.5            | 15分钟考核 |

### 4.2 偏差管理策略

```python
class DeviationOptimizer:
    """
    偏差优化管理器
    通过储能和柔性负荷最小化偏差考核费用
    """
    def __init__(self):
        self.precision = Decimal('0.000001')

    def calculate_deviation_cost(self,
                                   scheduled: Decimal,
                                   actual: Decimal,
                                   market_price: Decimal,
                                   tolerance_pct: Decimal = Decimal('3.0')) -> Dict:
        """
        计算偏差考核费用
        """
        deviation = actual - scheduled
        deviation_pct = (deviation / scheduled * 100) if scheduled > 0 else Decimal('0')

        # 免考核范围
        free_tolerance = scheduled * tolerance_pct / Decimal('100')

        if abs(deviation) <= free_tolerance:
            # 在免考核范围内
            penalty = Decimal('0')
            status = "WITHIN_TOLERANCE"
        else:
            # 超出免考核范围
            excess = abs(deviation) - free_tolerance

            if deviation > 0:  # 超发
                penalty = excess * market_price * Decimal('0.8')
                status = "OVER_GENERATION"
            else:  # 欠发
                penalty = excess * market_price * Decimal('1.2')
                status = "UNDER_GENERATION"

        return {
            "scheduled_mwh": scheduled.quantize(self.precision),
            "actual_mwh": actual.quantize(self.precision),
            "deviation_mwh": deviation.quantize(self.precision),
            "deviation_pct": deviation_pct.quantize(Decimal('0.01')),
            "free_tolerance_mwh": free_tolerance.quantize(self.precision),
            "status": status,
            "penalty_cny": penalty.quantize(self.precision),
            "market_price": market_price.quantize(self.precision)
        }

    def optimize_battery_for_deviation(self,
                                         forecast_error_pct: float,
                                         battery_capacity_mwh: float,
                                         battery_power_mw: float) -> Dict:
        """
        优化储能用于偏差管理
        """
        # 基于预测误差确定预留容量
        if forecast_error_pct < 2.0:
            reserve_ratio = 0.02  # 高精度预测
        elif forecast_error_pct < 5.0:
            reserve_ratio = 0.05  # 中等精度
        else:
            reserve_ratio = 0.10  # 低精度

        reserve_energy = battery_capacity_mwh * reserve_ratio
        reserve_power = min(battery_power_mw, reserve_energy * 4)  # 15分钟响应

        return {
            "forecast_error_pct": forecast_error_pct,
            "recommended_reserve_mwh": reserve_energy,
            "recommended_reserve_mw": reserve_power,
            "strategy": "Use battery for fast deviation correction within 15-min settlement period",
            "estimated_penalty_saving_pct": reserve_ratio * 100
        }

    def calculate_daily_deviation_summary(self,
                                            periods: List[Dict]) -> Dict:
        """
        计算日偏差汇总
        """
        total_penalty = Decimal('0')
        total_energy = Decimal('0')
        total_deviation = Decimal('0')
        within_tolerance_count = 0
        over_count = 0
        under_count = 0

        for period in periods:
            result = self.calculate_deviation_cost(
                Decimal(str(period['scheduled'])),
                Decimal(str(period['actual'])),
                Decimal(str(period['market_price']))
            )

            total_penalty += result['penalty_cny']
            total_energy += result['actual_mwh']
            total_deviation += abs(result['deviation_mwh'])

            if result['status'] == 'WITHIN_TOLERANCE':
                within_tolerance_count += 1
            elif result['status'] == 'OVER_GENERATION':
                over_count += 1
            else:
                under_count += 1

        return {
            "total_penalty_cny": total_penalty.quantize(self.precision),
            "total_energy_mwh": total_energy.quantize(self.precision),
            "total_abs_deviation_mwh": total_deviation.quantize(self.precision),
            "avg_deviation_pct": (total_deviation / total_energy * 100).quantize(Decimal('0.01')) if total_energy > 0 else Decimal('0'),
            "within_tolerance_count": within_tolerance_count,
            "over_generation_count": over_count,
            "under_generation_count": under_count,
            "total_periods": len(periods),
            "compliance_rate": (within_tolerance_count / len(periods) * 100) if periods else 0
        }
```

## 5. VPP收益分配

### 5.1 收益分配模型

VPP平台与业主之间的收益分配基于合同约定的分成比例：

```
收益分配公式：

总收益 = 现货市场收益 + 中长期合同收益 + 辅助服务收益 + 绿电溢价

VPP服务费 = 总收益 × 服务费率（通常5-15%）

业主收益 = 总收益 - VPP服务费 - 偏差考核费用

业主单位收益 = 业主收益 / 业主贡献电量（元/MWh）
```

### 5.2 收益分配引擎

```python
class VPPProfitDistribution:
    """
    VPP收益分配引擎
    """
    def __init__(self):
        self.precision = Decimal('0.000001')

    def calculate_profit_distribution(self,
                                        period_id: int,
                                        owners: List[Dict],
                                        total_revenue: Decimal,
                                        total_deviation_penalty: Decimal,
                                        vpp_service_rate: Decimal = Decimal('0.10')) -> List[Dict]:
        """
        计算VPP收益分配
        """
        # VPP服务费
        vpp_fee = total_revenue * vpp_service_rate

        # 可分配收益
        distributable = total_revenue - vpp_fee - total_deviation_penalty

        # 计算各业主贡献占比
        total_contribution = sum(
            Decimal(str(o['energy_contributed_mwh'])) for o in owners
        )

        distributions = []
        for owner in owners:
            contribution = Decimal(str(owner['energy_contributed_mwh']))
            contribution_ratio = contribution / total_contribution if total_contribution > 0 else Decimal('0')

            # 基础收益（按贡献比例）
            base_profit = distributable * contribution_ratio

            # 灵活性奖励（基于响应速度）
            flexibility_score = Decimal(str(owner.get('flexibility_score', 0.5)))
            bonus = base_profit * flexibility_score * Decimal('0.1')  # 最多10%奖励

            # 业主最终收益
            owner_profit = base_profit + bonus

            distributions.append({
                "owner_id": owner['owner_id'],
                "owner_name": owner.get('owner_name', ''),
                "energy_contributed_mwh": contribution.quantize(self.precision),
                "contribution_ratio_pct": (contribution_ratio * 100).quantize(Decimal('0.01')),
                "flexibility_score": flexibility_score.quantize(Decimal('0.01')),
                "base_profit_cny": base_profit.quantize(self.precision),
                "flexibility_bonus_cny": bonus.quantize(self.precision),
                "total_profit_cny": owner_profit.quantize(self.precision),

            "distributable_profit": distributable.quantize(self.precision),

            "owner_count": len(owners),
```
