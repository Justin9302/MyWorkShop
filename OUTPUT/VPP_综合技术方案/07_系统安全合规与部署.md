### 4.2 边缘安全逻辑

```java
// 边缘本地安全逻辑（OpenEMS插件）
// 参考平台推进方案中的安全逻辑

public class EdgeSafetyController {
    private static final int CLOUD_TIMEOUT_SECONDS = 600;
    private static final double BATTERY_SOC_MIN = 10.0;
    private static final double BATTERY_TEMP_MAX = 55.0;

    /**
     * 云端指令过期检测
     * 超过10分钟未收到云端指令，降级为本地安全模式
     */
    public void checkCloudCommandAge(long lastCloudCommandTime) {
        long ageSeconds = (System.currentTimeMillis() - lastCloudCommandTime) / 1000;
        if (ageSeconds > CLOUD_TIMEOUT_SECONDS) {
            // 云端指令过期，执行本地安全模式
            executeLocalSafeMode();
        }
    }

    /**
     * 硬件层强制保护
     * 无视云端指令，直接执行硬件保护
     */
    public void checkHardwareProtection(double batterySoc, double batteryTemp) {
        if (batterySoc < BATTERY_SOC_MIN || batteryTemp > BATTERY_TEMP_MAX) {
            // 硬件层强制保护
            forceStopControl();
        }
    }

    /**
     * 本地安全模式
     * 降级为自发自用模式，拒绝套利
     */
    private void executeLocalSafeMode() {
        // 1. 停止所有套利策略
        stopAllArbitrageStrategies();

        // 2. 切换为自发自用模式
        setOperationMode("SELF_CONSUMPTION");

        // 3. 记录SOE事件
        recordSOEEvent("LOCAL_SAFE_MODE_ACTIVATED",
            "Cloud command timeout exceeded " + CLOUD_TIMEOUT_SECONDS + "s");

        // 4. 通知运维人员
        notifyOperator("System entered local safe mode");
    }

    /**
     * 强制停止控制
     */
    private void forceStopControl() {
        // 1. 立即停止所有功率输出
        setAllPowerOutputs(0);

        // 2. 断开并网开关
        disconnectGridSwitch();

        // 3. 记录SOE事件
        recordSOEEvent("FORCE_STOP",
            "Battery SOC=" + BATTERY_SOC_MIN + "% or Temp=" + BATTERY_TEMP_MAX + "°C");

        // 4. 触发硬件告警
        triggerHardwareAlarm();
    }
}
```

### 4.3 联络线安全约束

```python
# 联络线约束示例（Pyomo）
# 参考平台推进方案中的联络线约束逻辑

def tie_line_constraints(model):
    """
    联络线物理约束
    """
    # 约束1：跨省购电量 <= 联络线剩余可用容量(ATC)
    def tie_line_rule(model, t):
        return model.Gansu_Import[t] <= model.Tie_Line_ATC[t]
    model.atc_cons = Constraint(model.Time, rule=tie_line_rule)

    # 约束2：四川保供电预警时，自动挂起套利
    def emergency_mode_rule(model, t):
        if model.Sichuan_Emergency[t] == 1:
            return model.Arbitrage_Active[t] == 0
        return Constraint.Skip
    model.emergency_cons = Constraint(model.Time, rule=emergency_mode_rule)

    # 约束3：总消费电量平衡
    def power_balance_rule(model, t):
        return (model.Local_Load[t] ==
                model.Local_PV[t] + model.Local_Hydro[t] +
                model.Sichuan_Purchase[t] + model.Gansu_Import[t] +
                model.Battery_Discharge[t] - model.Battery_Charge[t])
    model.balance_cons = Constraint(model.Time, rule=power_balance_rule)

    return model
```
