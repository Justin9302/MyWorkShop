### 2.1 边缘安全逻辑（Java/OpenEMS）

```java
// 边缘本地安全逻辑
// 参考平台推进方案中的安全逻辑

public class EdgeSafetyController {
    private static final int CLOUD_TIMEOUT_SECONDS = 600;
    private static final double BATTERY_SOC_MIN = 10.0;
    private static final double BATTERY_TEMP_MAX = 55.0;

    public void checkCloudCommandAge(long lastCloudCommandTime) {
        long ageSeconds = (System.currentTimeMillis() - lastCloudCommandTime) / 1000;
        if (ageSeconds > CLOUD_TIMEOUT_SECONDS) {
            executeLocalSafeMode();
        }
    }

    public void checkHardwareProtection(double batterySoc, double batteryTemp) {
        if (batterySoc < BATTERY_SOC_MIN || batteryTemp > BATTERY_TEMP_MAX) {
            forceStopControl();
        }
    }

    private void executeLocalSafeMode() {
        stopAllArbitrageStrategies();
        setOperationMode("SELF_CONSUMPTION");
        recordSOEEvent("LOCAL_SAFE_MODE_ACTIVATED",
            "Cloud command timeout exceeded " + CLOUD_TIMEOUT_SECONDS + "s");
        notifyOperator("System entered local safe mode");
    }

    private void forceStopControl() {
        setAllPowerOutputs(0);
        disconnectGridSwitch();
        recordSOEEvent("FORCE_STOP",
            "Battery SOC=" + BATTERY_SOC_MIN + "% or Temp=" + BATTERY_TEMP_MAX + "°C");
        triggerHardwareAlarm();
    }
}
```

### 2.2 联络线约束（Python/Pyomo）

```python
# 联络线约束示例
# 参考平台推进方案中的联络线约束逻辑

def tie_line_constraints(model):
    """联络线物理约束"""
    # 约束1：跨省购电量 <= 联络线剩余可用容量(ATC)
    def tie_line_rule(model, t):
        return model.Gansu_Import[t] <= model.Tie_Line_ATC[t]
    model.atc_cons = Constraint(model.Time, rule=tie_line_rule)

    # 约束2：四川保供电预警时，自动挂起套利
    def emergency_mode_rule(model, t):
        if model.Sichuan_Emergency[t] == 1:
            return model.Arbitrage_Active[t] == 0
        return Constraint.Skip
    model.emergency_cons = Constraint(model.Time, rule=emergency_mode_rule)

    # 约束3：功率平衡
    def power_balance_rule(model, t):
        return (model.Local_Load[t] ==
                model.Local_PV[t] + model.Local_Hydro[t] +
                model.Sichuan_Purchase[t] + model.Gansu_Import[t] +
                model.Battery_Discharge[t] - model.Battery_Charge[t])
    model.balance_cons = Constraint(model.Time, rule=power_balance_rule)

    return model
```

### 2.3 电价预测（Python/epftoolbox）

```python
# 电价预测示例
# 使用epftoolbox进行现货电价预测

from epftoolbox.models import CNN
from epftoolbox.data import read_data
import numpy as np

def predict_da_price(market="SICHUAN", date="2026-05-10"):
    """
    预测日前电价
    """
    # 读取历史数据
    df = read_data(dataset=market, years=2)

    # 训练CNN模型
    model = CNN()
    model.fit(df)

    # 预测未来24小时电价
    forecast = model.predict_online(df, date=date)

    return forecast

def predict_price_spread(market_a="SICHUAN", market_b="GANSU"):
    """
    预测省间价差
    """
    price_a = predict_da_price(market_a)
    price_b = predict_da_price(market_b)
    spread = price_a - price_b
    return spread
```

### 2.4 报价曲线优化（Python/Pyomo）

```python
# 报价曲线优化示例

def optimize_bidding_curve(price_forecast, battery_params, contract_params):
    """
    计算最优报价曲线
    """
    model = ConcreteModel()

    # 时段定义
    T = range(96)  # 24小时 × 15分钟
    model.Time = Set(initialize=T)

    # 变量
    model.Battery_Charge = Var(model.Time, within=NonNegativeReals)
    model.Battery_Discharge = Var(model.Time, within=NonNegativeReals)
    model.SOC = Var(model.Time, within=NonNegativeReals)
    model.Grid_Import = Var(model.Time, within=NonNegativeReals)
    model.Grid_Export = Var(model.Time, within=NonNegativeReals)

    # 目标：最大化收益
    def objective_rule(model):
        revenue = sum(
            price_forecast[t] * (model.Grid_Export[t] - model.Grid_Import[t])
            for t in model.Time
        )
        return revenue
    model.Objective = Objective(rule=objective_rule, sense=maximize)

    # 约束
    def soc_balance_rule(model, t):
        if t == 0:
            return model.SOC[t] == battery_params['initial_soc']
        return model.SOC[t] == (model.SOC[t-1] +
            model.Battery_Charge[t] * battery_params['efficiency'] -
            model.Battery_Discharge[t] / battery_params['efficiency'])
    model.soc_balance = Constraint(model.Time, rule=soc_balance_rule)

    # 求解
    solver = SolverFactory('scip')
    solver.solve(model)

    # 生成报价曲线
    bidding_curve = []
    for t in T:
        net_position = model.Grid_Export[t].value - model.Grid_Import[t].value
        bidding_curve.append({
            "period": t,
            "price": price_forecast[t],
            "quantity": net_position
        })

    return bidding_curve
```

### 2.5 结算精度控制（Python/Decimal）

```python
# 财务精度控制示例

from decimal import Decimal, ROUND_HALF_UP

class PrecisionController:
    """财务精度控制器"""
    def __init__(self):
        self.precision = Decimal('0.000001')
        self.rounding = ROUND_HALF_UP

    def round_amount(self, value: Decimal) -> Decimal:
        return value.quantize(self.precision, rounding=self.rounding)

    def sum_with_precision(self, values: list) -> Decimal:
        total = sum(values, Decimal('0'))
        return self.round_amount(total)

    def multiply_with_precision(self, a: Decimal, b: Decimal) -> Decimal:
        return self.round_amount(a * b)
```
